\section{正则空间上的等价关系}

记号约定:$X$是正则空间,$R$是$X$上的等价关系.

\begin{proposition}{正则空间上的闭等价关系的图也是闭的}{}
	若$R$是闭等价关系,则相对$R$的二元关系$C$是$X\times X$的闭集.
\end{proposition}

\begin{corollary}{既开又闭等价关系的Hausdorff性}{}
	如果$R$既是开等价关系又是闭等价关系,那么$R$是Hausdorff等价关系.
\end{corollary}

\begin{proposition}{闭子集坍缩为商空间的Hausdorff性}{}
	设$F$是$X$中的闭集.若$R$通过把$F$中所有点等同起来以后得到,则$X/R$是Hausdorff空间.
\end{proposition}
