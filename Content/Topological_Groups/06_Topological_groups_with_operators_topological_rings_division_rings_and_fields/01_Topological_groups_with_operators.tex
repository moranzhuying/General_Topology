\chapter{拓扑\texorpdfstring{$\Omega$}{Omega}-群,拓扑除环,拓扑域}

\section{拓扑\texorpdfstring{$\Omega$}{Omega}-群}

记号约定:$\Omega$是集合,$G$是$\Omega$-群,它作为群是拓扑群.

\begin{definition}{拓扑$\Omega$-群}{}
	如果对每个$\alpha\in\Omega$,它诱导的群同态$f_{\alpha}:G\to G$连续,则称$G$是拓扑$\Omega$-群.
\end{definition}

\begin{proposition}{$\Omega$-群的$\Omega$-不变子群(及其子群)是拓扑$\Omega$-群}{}
	设$G$是拓扑$\Omega$-群,$H$是$G$的$\Omega$-不变子群.
	\begin{enumerate}
		\item $H$是拓扑$\Omega$-不变子群.
		\item $\closure\left(H\right)$是$G$的$\Omega$-不变子群
	\end{enumerate}
\end{proposition}

\begin{proposition}{拓扑$\Omega$-群的商群是拓扑$\Omega$-群}{}
	设$G$是拓扑$\Omega$-群,$H$是$G$的$\Omega$-正规子群.给$G/H$装备商拓扑,那么$G/H$是拓扑$\Omega$-群.
\end{proposition}

\begin{proposition}{拓扑$\Omega$-群的乘积是拓扑$\Omega$-群}{}
	设$\left(G_{i}\right)_{i\in I}$是一族拓扑$\Omega$-群.给$\prod\limits_{i\in I}G_{i}$装备成绩拓扑,那么它是拓扑$\Omega$-群.
\end{proposition}

\begin{proposition}{}{}
	设$G$是Hausdorff的拓扑$\Omega$-群.如果$G$可Hausdorff完备化,那么$G$的Hausdorff完备化是拓扑$\Omega$-群.
\end{proposition}
