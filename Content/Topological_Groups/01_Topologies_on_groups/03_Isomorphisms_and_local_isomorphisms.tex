\section{拓扑群同构,局部同构}

记号约定:$\left(G,\cdot,e\right)$和$\left(G^{\prime},\cdot^{\prime},e^{\prime}\right)$是拓扑群.

\begin{definition}{拓扑群同态,拓扑群同构}{}
	设$f\in\Hom_{\mathsf{Set}}\left(X,Y\right)$.
	\begin{enumerate}
		\item 若$f$既是群同态又是连续映射,则称$f$是拓扑群同态.
		\item 若$f$既是群同构又是同胚,则称$f$是拓扑群同构.
		\item 记$\Hom_{\mathsf{TopGrp}}\left(G,G^{\prime}\right)$是$G$到$G^{\prime}$的拓扑群同态构成的集合,$\End_{\mathsf{TopGrp}}\left(G\right)=\Hom_{\mathsf{TopGrp}}\left(G,G\right)$.
		\item 记$\Isom_{\mathsf{TopGrp}}\left(G,G^{\prime}\right)$是$G$到$G^{\prime}$的拓扑群同态构成的集合,$\Aut_{\mathsf{TopGrp}}\left(G\right)=\Isom_{\mathsf{TopGrp}}\left(G,G\right)$.
	\end{enumerate}
\end{definition}

\begin{definition}{拓扑群的局部同构}{}
	\begin{enumerate}
		\item 设$V\in\mathcal{N}\left(e\right),V^{\prime}\in\mathcal{N}\left(e^{\prime}\right),f\in\Hom_{\mathsf{Top}}\left(V,V^{\prime}\right)$.如果
		\begin{enumerate}
			\item $\forall x,y\in V\left(xy\in V\implies f\left(xy\right)=f\left(x\right)f\left(y\right)\right)$,
			\item $\forall x^{\prime},y^{\prime}\in V^{\prime}\left(x^{\prime}y^{\prime}\in V^{\prime}\implies f^{-1}\left(x^{\prime}y^{\prime}\right)=f^{-1}\left(x^{\prime}\right)f^{-1}\left(y^{\prime}\right)\right)$,
		\end{enumerate}
		则称$f$是$G$到$G^{\prime}$的局部同构.特别地,当$G=G^{\prime}$时,称$f$是局部自同构.
		\item 如果存在局部同构$G\to G^{\prime}$,则称$G$和$G^{\prime}$局部同构.
	\end{enumerate}
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 拓扑群同构$\implies$局部同构,反向的蕴含不成立.例如,拓扑群$\R$局部同构于$\mathbb{T}$,但是二者作为拓扑群并不同构.
		\item 如果$f$是局部同构,那么对每个$W\in\mathcal{N}\left(e\right)$,映射$f|_{V\cap W}$也是局部同构.
		\item 一般来说,如果$f$只满足(a),那么它不一定自动满足(b).
	\end{enumerate}
\end{remark}

\begin{proposition}{局部同构可以延拓}{}
	设$V\in\mathcal{N}\left(e\right),V^{\prime}\in\mathcal{N}\left(e^{\prime}\right),f\in\Hom_{\mathsf{Top}}\left(V,V^{\prime}\right)$.如果
	\begin{itemize}
		\item $\forall x,y\in V\left(xy\in V\implies f\left(xy\right)=f\left(x\right)f\left(y\right)\right)$,
		\item $W\in\mathcal{N}\left(e\right)$且$WW\subseteq V$,
	\end{itemize}
	那么存在局部同构$g:W\to V^{\prime}$使得下图交换.
	\begin{center}
		\begin{tikzcd}
			V && {V^{\prime}} \\
			& W
			\arrow["f", from=1-1, to=1-3]
			\arrow[hook', from=2-2, to=1-1]
			\arrow["{\exists g}"', dashed, from=2-2, to=1-3]
		\end{tikzcd}
	\end{center}
\end{proposition}
