\section{拓扑群的连通分支}

未加声明时,$G$是拓扑群(乘法记号).

\begin{definition}{对称邻域生成的子群}{}
	设$V$是$e$的对称邻域.记$V^{\infty}$为$V$生成的子群.
\end{definition}

\begin{remark}{}{}
	$V^{\infty}$是$G$的开子群和闭子群.
\end{remark}

\begin{proposition}{连通拓扑群的生成元}{}
	如果$G$是连通的,那么$e$的每个对称邻域$V$都生成$G$.
\end{proposition}

\begin{proposition}{连通集生成的子群也连通}{}
	设$M$是$G$的包含$e$的子集.如果$M$连通,那么$M$生成的子群也连通.
\end{proposition}

\begin{definition}{单位连通分支}{}
	我们称$G$中包含$e$的连通分支是$G$的单位元分支(或恒等分支),记作$G^{0}$.
\end{definition}

\begin{proposition}{单位连通分支的性质}{}
	设$K$是$G$的单位连通分支.
	\begin{enumerate}
		\item $K$是$G$的闭正规子群.
		\item 对每个$x\in G$,它在$G$中的连通分支是$xK$,并且$xK=Kx$.
	\end{enumerate}
\end{proposition}
