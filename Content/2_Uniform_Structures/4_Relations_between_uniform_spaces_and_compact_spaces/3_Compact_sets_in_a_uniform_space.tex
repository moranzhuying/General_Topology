\section{一致空间中的紧集}

记号约定:$X$是一致空间,其一致结构为$\mathfrak{U}$.

\begin{proposition}{紧集和闭集的一致分离}{}
	设$A$是$X$的紧集,$B$是$X$的闭集,$A\cap B=\emptyset$,则存在$V\in\mathfrak{U}$使得$V\left(A\right)\cap V\left(B\right)=\emptyset$.
\end{proposition}

\begin{corollary}{紧集的邻域基}{}
	设$A$是$X$的紧集,则$\left\{V\left(A\right)\middle|A\in\mathfrak{U}\right\}$是$A$在$X$中的邻域基.
\end{corollary}
