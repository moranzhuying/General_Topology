\section{\texorpdfstring{$\R^{n}$}{Rn}的Hausdorff商群}

\begin{definition}
	我们称$\T^{n}\coloneq\left(\R/\Z\right)^{n}$是$n$维环面.作为拓扑群$\T^{n}$同构于$\R^{n}/\Z^{n}$.
\end{definition}

\begin{theorem}
	设$H$是$\R^{n}$的闭子群,则存在$h,k\in\Z_{\geqslant 0}$且$h+k\leqslant n$使得$\R^{n}/H$作为拓扑群同构于$\R^{h}\times\T^{k}$.
\end{theorem}

\begin{proposition}
	$\T^{n}$是紧致,连通且局部连通的拓扑空间.
\end{proposition}

\begin{proposition}
	设$C=\left[0,1\right]^{n}\subseteq\R^{n}$.定义$C$上的等价关系$\sim$:
	\begin{align*}
		\forall\left(x_{1},\ldots,x_{n}\right),\left(y_{1},\ldots,y_{n}\right)\in C\left(\left(x_{1},\ldots,x_{n}\right)\sim\left(y_{1},\ldots,y_{n}\right)\iff\forall i\in\llbracket 1,n\rrbracket\left(x_{i}-y_{i}\in\Z\right)\right),
	\end{align*}
	则商空间$C/\sim$和$\T^{n}$同胚.
\end{proposition}

\begin{proposition}
	对任一$p\in\llbracket 0,n\rrbracket$,$\R^{p}\times\T^{n-p}$局部同构于$\R^{n}$.特别地,$\T^{n}$局部同构于$\R^{n}$.
\end{proposition}
