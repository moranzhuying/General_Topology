\section{\texorpdfstring{$\T^{n}$}{Tn}的子群和商群}

未加声明时,$G$是$\T^{n}$的子群,$\varphi:\R^{n}\twoheadrightarrow\T^{n}$是典范满射.

\begin{proposition}
	存在$\R^{n}$的子群$H$使得$\Z^{n}\subseteq H$且$G=\varphi\left(H\right)$.进一步:
	\begin{itemize}
		\item 作为群有同构$G\simeq H/\Z^{n}$;
		\item $G$是$\T^{n}$的闭子群$\iff$ $H$是$\R^{n}$的闭子群.
	\end{itemize}
\end{proposition}

\begin{theorem}
	设$G$是$\T^{n}$的闭子群,则存在$h\in\llbracket 0,n\rrbracket$,$r\in\llbracket 0,n-h\rrbracket$和$\Z_{\geqslant 1}$中的序列$\left(e_{i}\right)_{i=1}^{r}$满足如下条件:
	\begin{itemize}
		\item $\forall i\in\llbracket 1,r-1\rrbracket\left(e_{i}\mid e_{i+1}\right)$;
		\item $G$作为拓扑群同构于$\T^{n}\times\left(\Z/e_{1}\Z\times\ldots\times\Z/e_{r}\Z\right)$.
	\end{itemize}
\end{theorem}

\begin{theorem}
	设$G$是$\T^{n}$的闭子群,则存在$m\in\llbracket 0,n\rrbracket$使得$\T^{m}/G$作为拓扑群同构于$\T^{m}$.
\end{theorem}

\begin{corollary}
	设$H$是$\T$的闭子群,则下列情形之一成立:
	\begin{itemize}
		\item $G=\T$,并且$\T/G$作为拓扑群同构于$\left\{0\right\}$;
		\item 存在$m\in\Z_{\geqslant 1}$使得$G$作为拓扑群同构于配备离散拓扑的拓扑群$\Z/m\Z$,并且$\T/G$作为拓扑群同构于$\T$.
	\end{itemize}
\end{corollary}
