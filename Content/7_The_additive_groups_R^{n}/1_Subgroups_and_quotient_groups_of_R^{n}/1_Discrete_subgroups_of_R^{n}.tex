\chapter{\texorpdfstring{$\R^{n}$}{Rn}的子群和商群}

\section{\texorpdfstring{$\R^{n}$}{Rn}的离散子群}

没有特别声明时,$p\in\llbracket 1,n\rrbracket$.论及$\R^{n}$的离散子群$G$的秩,一律指将$G$视为$\Z$-模后$G$的维数.

\begin{proposition}
	设$\left(a_{i}\right)_{i=1}^{p}$是$\R^{n}$的$\R$-线性无关族,$G$是$\left(a_{i}\right)_{i=1}^{p}$生成的$\R^{n}$的子群.
	\begin{enumerate}
		\item $G$是$\R^{n}$的离散子群,作为拓扑群同构于$\Z^{p}$,其中$\Z$配备离散拓扑,$\Z^{p}$配备乘积拓扑.
		\item 存在拓扑模同态$f:\R^{n}\to\R^{n}$使得$f\left(G\right)=\Z^{p}$.
	\end{enumerate}
\end{proposition}

\begin{proposition}
	设$G$是$\R^{n}$的秩$p$离散子群,$\left(a_{i}\right)_{i=1}^{p}$是$G$中的$\R$-自由族,$P=\left\{\sum\limits_{i=1}^{n}t_{i}a_{i}\in\R\middle|\forall i\in\llbracket 1,p\rrbracket\left(t_{i}\in\left[0,1\right]\right)\right\}$.
	\begin{enumerate}
		\item $G\cap P$是有限集,并且$G$作为群的生成集是$G\cap P$.
		\item $\left(a_{i}\right)_{i=1}^{n}$是$G$的$\Q$-带标基.
	\end{enumerate}
\end{proposition}

\begin{corollary}
	设$\left(a_{i}\right)_{i=1}^{p}$是$\R^{n}$的$\R$-自由族,$\left(t_{i}\right)_{i=1}^{p}$是$\R$中的序列,$b=\sum\limits_{i=1}^{n}t_{i}a_{i}$.下列命题等价:
	\begin{enumerate}
		\item $\left\{a_{1},\ldots,a_{n},b\right\}$生成的子群是离散群.
		\item $\forall i\in\llbracket 1,p\rrbracket\left(t_{i}\in\Q\right)$.
	\end{enumerate}
\end{corollary}

\begin{theorem}
	设$\left(\theta_{i}\right)_{i=1}^{n}$是$\R$的序列.对任一$\varepsilon\in\R_{>0}$,存在$q\in\Z$和$\Z$中的序列$\left(p_{i}\right)_{i=1}^{n}$使得
	\begin{align*}
		\forall i\in\llbracket 1,n\rrbracket\left(\left|q\theta_{i}-p_{i}\right|\leqslant\varepsilon\right),
	\end{align*}
	并且$\exists j\in\llbracket 1,n\rrbracket\left(\left|q\theta_{i}-p_{i}\right|>\varepsilon\right)\iff\exists k\in\llbracket 1,n\rrbracket\left(\theta_{i}\in\R\setminus\Q\right)$.
\end{theorem}

\begin{theorem}
	$\R^{n}$的每个秩$p$的离散子群作为拓扑群和$\Z^{p}$同构.
\end{theorem}

\begin{definition}
	$\R^{n}$的秩$n$离散子群称为$\R^{n}$中的格.
\end{definition}
