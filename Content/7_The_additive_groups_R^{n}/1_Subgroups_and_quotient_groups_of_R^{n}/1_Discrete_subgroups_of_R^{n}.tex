\chapter{\texorpdfstring{$\R^{n}$}{Rn}的子群和商群}

记号约定:$0\leq p\leq n$.

\begin{definition}{子集的秩和有理秩}{}
	设$A\subseteq\R^{n}$.我们分别称$A$生成的$\R$-线性子空间和$\Q$-线性子空间的维数为$A$的秩和有理秩.
\end{definition}

\begin{alarm}{}{}
	我们需要区分$A$生成的子群和$A$生成的$\R$-子空间,前者是$\sum\limits_{x\in A}\Z x$,后者是$\sum\limits_{x\in A}\R x$.此外,我们要注意线性相关(线性无关)依赖于基环的选择:如果$A$是$\Q$-线性无关的,那么$A$也是$\R$-线性无关的,反之不成立.
\end{alarm}

\section{\texorpdfstring{$\R^{n}$}{Rn}的离散子群}

\begin{example}{$\R^{n}$的$\Z^{p}$型子群}{}
	设$\left(\mathbf{a}_{i}\right)_{i=1}^{p}$是$\R^{n}$的线性无关族,那么$\langle\mathbf{a}_{1},\ldots,\mathbf{a}_{p}\rangle$是$\R^{n}$的同构于$\Z^{p}$的秩$p$离散子群.
\end{example}

\begin{remark}{}{}
	在\textit{经典代数学}中,我们已经研究过了PID上有限生成模的结构.如果$\left(\mathbf{b}_{i}\right)_{i=1}^{p}$是$\langle\mathbf{a}_{1},\ldots,\mathbf{a}_{p}\rangle$的基,那么从$\left(\mathbf{a}_{i}\right)_{i=1}^{p}$到$\left(\mathbf{b}_{i}\right)_{i=1}^{p}$的过渡矩阵是$\SL_{p}\left(\Z\right)$的元素.
\end{remark}

\begin{proposition}{}{}
	设$G$是$\R^{n}$的秩$p$离散子群,$\left(a_{i}\right)_{i=1}^{p}$是$\R^{n}$的线性无关族,$P=\sum\limits_{i=1}^{p}\left[-1,1\right]\mathbf{a}_{i}$.
	\begin{enumerate}
		\item $G\cap P$是有限集,并且生成$G$.
		\item $G\subseteq\sum\limits_{i=1}^{p}\Q\mathbf{a}_{i}$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	设$\left(\mathbf{a}_{i}\right)_{i=1}^{p}$是$\R^{n}$中的线性无关族,$\mathbf{b}=\sum\limits_{i=1}^{p}t_{i}\mathbf{a}_{i}\in\sum\limits_{i=1}^{p}\R\mathbf{a}_{i}$.下列陈述等价:
	\begin{enumerate}
		\item $\mathbf{a}_{1},\ldots,\mathbf{a}_{p},\mathbf{b}$生成的子群离散.
		\item $\left(t_{i}\right)_{i=1}^{p}$中的每一项都是有理数.
	\end{enumerate}
\end{corollary}

\begin{remark}{}{}
	由此我们有等价的描述:对于$\R^{n}$的离散子群$G$,如果其元素族$\left(\mathbf{x}_{i}\right)_{i=1}^{p}$是$\R$-线性相关的,那么$\left(\mathbf{x}_{i}\right)_{i=1}^{p}$也是$\Q$-线性相关的.由此可知,对$\R^{n}$的每个离散子群,它的有理秩和秩相等.
\end{remark}

\begin{proposition}{Kronecker定理}{}
	设$\left(\theta_{i}\right)_{i=1}^{n}$是$\R$中的序列.下列陈述等价:
	\begin{enumerate}
		\item 对每个$\varepsilon\in\R_{>0}$,存在$q\in\N$和$\N$中的序列$\left(p_{i}\right)_{i=1}^{n}$满足如下条件:
		\begin{itemize}
			\item 对每个$1\leq i\leq n$有$\left|q\theta_{i}-p_{i}\right|\leq\varepsilon$.
			\item 存在$1\leq j\leq n$使得$q\theta_{j}-p_{j}\neq 0$.
		\end{itemize}
		\item 存在$1\leq j\leq n$使得$\theta_{j}\in\R\setminus\Q$.
	\end{enumerate}
\end{proposition}

\begin{theorem}{}{}
	$\R^{n}$的每个秩$p$离散子群都可以由$\R^{n}$中$p$个线性无关的元素生成.
\end{theorem}

\begin{definition}{晶格}{}
	$\R^{n}$中的秩$n$离散子群称为$\R^{n}$中的晶格(或Bravais格).
\end{definition}
