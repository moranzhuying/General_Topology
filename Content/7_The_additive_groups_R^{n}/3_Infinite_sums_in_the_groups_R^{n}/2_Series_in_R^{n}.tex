\section{\texorpdfstring{$\R^{n}$}{Rn}中的级数}



未加声明时,$\left(x_{m}\right)_{m=0}^{\infty}$是$\R^{n}$中的序列.

\begin{proposition}{}{}
	设对任一$m\in\Z_{\geqslant 0}$,$x_{m}=\left(x_{m,1},\ldots,x_{m,n}\right)$,则$\sum\limits_{m=0}^{\infty}x_{m}$收敛$\iff$对任一$k\in\llbracket 1,n\rrbracket$,$\sum\limits_{m=0}^{\infty}x_{m,k}$收敛.
\end{proposition}

\begin{definition}{}{}
	若$\sum\limits_{m=0}^{\infty}\left\|x_{m}\right\|$收敛,则称$\sum\limits_{m=0}^{\infty}x_{m}$绝对收敛.
\end{definition}

\begin{proposition}{}{}
	$\sum\limits_{m=0}^{\infty}x_{m}$交换收敛$\iff$ $\sum\limits_{m=0}^{\infty}x_{m}$绝对收敛.
\end{proposition}
