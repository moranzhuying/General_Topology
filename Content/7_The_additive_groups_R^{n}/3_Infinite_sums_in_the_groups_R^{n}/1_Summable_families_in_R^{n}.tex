\chapter{群\texorpdfstring{$\R^{n}$}{Rn}中的无限和}

\section{\texorpdfstring{$\R^{n}$}{Rn}中的可和族}

未加声明时,$\left(x_{i}\right)_{i\in I}$是$\R^{n}$的元素族.

\begin{proposition}{}{}
	设$\forall i\in I\left(x_{i}=\left(x_{i,1},\ldots,x_{i,n}\right)\right)$,则$\left(x_{i}\right)_{i\in I}$在$\R^{n}$中可和$\iff$对任一$j\in\llbracket 1,n\rrbracket$,$\left(x_{i,k}\right)_{i\in I}$在$\R$中可和.
\end{proposition}

\begin{theorem}{}{}
	$\left(x_{i}\right)_{i\in I}$在$\R$中可和$\iff$ $\left(\left\|x_{i}\right\|\right)_{i\in I}$在$\R$中可和.
\end{theorem}

\begin{proposition}{}{}
	设$I$是有限集,则$\sum\limits_{i\in I}\left\|x_{i}\right\|\leqslant 2n\cdot\sup\limits_{J\subseteq I}\left\|\sum\limits_{i\in J}x_{i}\right\|$.
\end{proposition}

\begin{corollary}{}{}
	$\left(x_{i}\right)_{i\in I}$可和$\iff$ $\left\{\sum\limits_{i\in J}x_{i}\in\R^{n}\middle|J\in\fin\left(I\right)\right\}$在$\R^{n}$中有界.
\end{corollary}

未加声明时,$\left(r_{i}\right)_{i\in I}$,$\left(x_{i}\right)_{i\in I}$是$\R^{m}$的可和族,$\left(y_{j}\right)_{j\in J}$是$\R^{n}$的可和族.

\begin{proposition}{}{}
	设$f:\R^{m}\times\R^{n}\to\R^{p}$是$\R$-双线性映射,则$\left(f\left(x_{i},y_{j}\right)\right)_{\left(i,j\right)\in I\times J}$在$\R^{p}$中可和,并且
	\begin{align*}
		\sum\limits_{\left(i,j\right)\in I\times J}f\left(x_{i},y_{j}\right)=f\left(\sum\limits_{i\in I}x_{i},\sum\limits_{j\in J}y_{j}\right).
	\end{align*}
\end{proposition}

\begin{corollary}{}{}
	设$\left(r_{i}\right)_{i\in I}$是$\R$的可和族,则$\left(r_{i}y_{j}\right)_{\left(i,j\right)\in I\times J}$是$\R^{n}$的可和族,并且$\sum\limits_{\left(i,j\right)\in I\times J}r_{i}y_{j}=\sum\limits_{i\in I}r_{i}\sum\limits_{j\in J}y_{j}$.
\end{corollary}

\begin{corollary}{}{}
	设$\left(y_{j}\right)_{j\in J}$是$\R^{n}$的可和族,则$\left(\left(x_{i}\middle|y_{j}\right)\right)_{\left(i,j\right)\in I\times J}$是$\R$上的可和族,并且$\sum\limits_{\left(i,j\right)\in I\times J}=\left(\sum\limits_{i\in I}x_{i}\middle|\sum\limits_{j\in J}y_{j}\right)$.
\end{corollary}
