\section{\texorpdfstring{$\R^{n}$}{Rn}到拓扑群的连续同态的局部定义}

未加声明时,$G$是乘法拓扑群.

\begin{proposition}
	设$A\subseteq\R^{n}$是包含原点的平行多面体,$f:A\to G$连续且
	\begin{align*}
		\forall x,y\in A\left(x+y\in A\implies f\left(x+y\right)=f\left(x\right)f\left(y\right)\right),
	\end{align*}
	则$f$能唯一地延拓为拓扑群同态$\bar{f}:\R^{n}\to G$.
\end{proposition}

\begin{corollary}
	设$V\subseteq\R^{n}$是$0$的邻域,$f:A\to G$连续且
	\begin{align*}
		\forall x,y\in A\left(x+y\in A\implies f\left(x+y\right)=f\left(x\right)f\left(y\right)\right),
	\end{align*}
	则存在唯一的拓扑群同态$\bar{f}:\R^{n}\to G$,存在$0$在$\R^{n}$中的邻域$U$使得$\forall x\in\R^{n}\left(\bar{f}\left(x\right)=f\left(x\right)\right)$.
\end{corollary}

\begin{corollary}
	设$f:\R^{n}\to G$是局部同构,则存在唯一的严格态射$\bar{f}:\R^{n}\to G$,存在$0$在$\R^{n}$中的邻域$U$使得
	\begin{align*}
		\forall x\in\R^{n}\left(\bar{f}\left(x\right)=f\left(x\right)\right),
	\end{align*}
	并且$\im\left(\bar{f}\right)$是$G$的开子群.
\end{corollary}

\begin{theorem}
	设$G$连通却局部同构于$\R^{n}$,则存在$p\in\llbracket 0,n\rrbracket$使得$G$同构于$\R^{p}\times\T^{n-p}$.
\end{theorem}
