\section{\texorpdfstring{$\T^{n}$}{Tn}的自同构}

\begin{theorem}
	设$H$是$\R^{n}$的闭子群,$\pi:\R^{n}\twoheadrightarrow\R^{n}/H$是典范满射.记
	\begin{align*}
		W_{1}=\left\{f\in G^{\R^{n}/H}\middle|f\text{是拓扑群同态}\right\},W_{2}=\left\{g\in W_{1}\middle|\forall h\in H\forall x\in\R^{n}\left(g\left(x+h\right)=g\left(x\right)\right)\right\}.
	\end{align*}
	则$\varphi:W_{1}\to W_{2},f\mapsto f\circ\pi$是可逆映射.
\end{theorem}

\begin{corollary}
	设$G$是\text{Hausdorff}拓扑群,$f:\T^{n}\to G$是拓扑群同态.
	\begin{enumerate}
		\item $f$是严格拓扑群同态,$\im\left(f\right)$是$G$的紧子群.
		\item 存在$p\in\llbracket 0,n\rrbracket$,使得$\im\left(f\right)$作为拓扑群同构于$\T^{p}$.
	\end{enumerate}
\end{corollary}

\begin{corollary}
	设$f:\T^{n}\to\R^{m}$是拓扑群同态,则$f$是平凡群同态.
\end{corollary}

\begin{proposition}
	设$f:\T^{n}\to\T^{p}$是拓扑群同态,则存在唯一的$\R$-模同态$u:\R^{n}\to\R^{p}$使下图交换($\pi_{n}$,$\pi_{p}$是典范满射):
	\begin{center}
		\begin{tikzcd}
			\R^{n} \arrow[d, "\pi_{n}"', two heads] \arrow[r, "\exists!u", dashed] & \R^{p} \arrow[d, "\pi_{p}", two heads] \\
			\T^{n} \arrow[r, "f"']                                                 & \T^{p}                                
		\end{tikzcd}
	\end{center}
	此外,$u\left(\Z^{n}\right)\subseteq\Z^{n}$.
\end{proposition}

\begin{proposition}
	设$u:\R^{n}\to\R^{p}$是$\R$-模同态,$u\left(\Z^{n}\right)\subseteq\Z^{n}$,则使下图交换的拓扑群同态$f$存在且唯一:
	\begin{center}
		\begin{tikzcd}
			\R^{n} \arrow[d, "\pi_{n}"', two heads] \arrow[r, "u"] & \R^{p} \arrow[d, "\pi_{p}", two heads] \\
			\T^{n} \arrow[r, "\exists!f"', dashed]                 & \T^{p}                                
		\end{tikzcd}
	\end{center}
	其中,$\pi_{n}$,$\pi_{p}$是典范满射.
\end{proposition}

\begin{proposition}
	设$\R$-模同态$u:\R^{n}\to\R^{p}$满足$u\left(\Z^{n}\right)\subseteq\Z^{p}$,拓扑群同态$f:\T^{n}\to\T^{p}$由$u$诱导.下列命题等价:
	\begin{enumerate}
		\item $f$是单射.
		\item $u$是单射,$\Z^{p}/u\left(\Z^{n}\right)$作为$\Z$-模无挠.
	\end{enumerate}
	上述条件之一成立时,有拓扑群同构$\im\left(f\right)\simeq\im\left(u\right)/u\left(\Z^{n}\right)$.
\end{proposition}

\begin{proposition}
	设$f:\T^{n}\to\T^{n}$是拓扑群同构,则存在唯一的$\R$-模同构$u:\R^{n}\to\R^{n}$使下图交换,并且$u\left(\Z^{n}\right)=\Z^{n}$:
	\begin{center}
		\begin{tikzcd}
			\R^{n} \arrow[d, "\pi_{n}"', two heads] \arrow[r, "\exists!u", dashed] & \R^{n} \arrow[d, "\pi_{n}", two heads] \\
			\T^{n} \arrow[r, "f"']                                                 & \T^{n}                                
		\end{tikzcd}
	\end{center}
	其中,$\pi_{n}$是典范满射.
\end{proposition}

\begin{proposition}
	设$u:\R^{n}\to\R^{n}$是$\R$-模自同构,$u\left(\Z^{n}\right)=\Z^{n}$,$\mathcal{B}\coloneq\left(e_{i}\right)_{i=1}^{n}$是$\R^{n}$的标准带标基,则
	\begin{align*}
		\left[u\right]_{\mathcal{B}\leftarrow\mathcal{B}}\in\mathbf{M}_{n}\left(\Z\right),\det\left[u\right]_{\mathcal{B}\leftarrow\mathcal{B}}\in\left\{-1,1\right\}.
	\end{align*}
\end{proposition}

\begin{proposition}
	设$u:\R^{n}\to\R^{n}$是$\R$-模自同构,$u\left(\Z^{n}\right)=\Z^{n}$,则存在唯一的拓扑群自同构$f:\T^{n}\to\T^{n}$使下图交换:
	\begin{center}
		\begin{tikzcd}
			\R^{n} \arrow[d, "\varphi"', two heads] \arrow[r, "u"] & \R^{n} \arrow[d, "\varphi", two heads] \\
			\T^{n} \arrow[r, "\exists!f"', dashed]                 & \T^{n}                                
		\end{tikzcd}
	\end{center}
	图中,$\pi$是典范满射.进一步,$\T^{n}$的自同构群同构于$\GL_{n}\left(\Z\right)$.
\end{proposition}

\begin{corollary}
	$\T$上的拓扑群自同构只有$\id_{\T}$和$-\id_{\T}$.
\end{corollary}
