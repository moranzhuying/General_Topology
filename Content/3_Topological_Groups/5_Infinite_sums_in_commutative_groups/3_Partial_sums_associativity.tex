\section{部分和的结合性}

记号约定:$G$是Hausdorff的交换拓扑群(加法记号),$\left(x_{i}\right)_{i\in I}$是$G$的元素族.

\begin{proposition}{完备群中可和族的子族遗传可和性}{}
	设$G$完备.如果$\left(x_{i}\right)_{i\in I}$可和,那么$\left(x_{i}\right)_{i\in I}$的每个子族都可和.
\end{proposition}

\begin{theorem}{完备拓扑群中求和的Fubini定理}{}
	设$G$完备,$\left(I_{\lambda}\right)_{\lambda\in\Lambda}$是$I$的划分,对每个$\lambda\in\Lambda$有$s_{\lambda}=\sum\limits_{i\in I_{\lambda}}x_{i}$.如果$\left(x_{i}\right)_{i\in I}$可和,那么$\left(s_{\lambda}\right)_{\lambda\in\Lambda}$可和,并且
	\begin{align*}
		\sum\limits_{i\in I}x_{i}=\sum\limits_{\lambda\in\Lambda}s_{\lambda}.
	\end{align*}
\end{theorem}

\begin{corollary}{}{}
	设$G$完备,$\left(x_{ij}\right)_{i\in I,j\in J}$是$G$的可和族,则$\sum\limits_{\left(i,j\right)\in I\times J}x_{ij}=\sum\limits_{i\in I}\sum\limits_{j\in J}x_{ij}=\sum\limits_{j\in J}\sum\limits_{i\in I}x_{ij}$.
\end{corollary}

\begin{proposition}{}{}
	设$\left(I_{\lambda}\right)_{\lambda\in\Lambda}$是$I$的有限划分.若对任一$\lambda\in\Lambda$,$\left(x_{i}\right)_{i\in I_{\lambda}}$可和,则$\left(x_{i}\right)_{i\in I}$可和,并且$\sum\limits_{i\in I}x_{i}=\sum\limits_{\lambda\in\Lambda}\sum\limits_{i\in I_{\lambda}}x_{i}$.
\end{proposition}
