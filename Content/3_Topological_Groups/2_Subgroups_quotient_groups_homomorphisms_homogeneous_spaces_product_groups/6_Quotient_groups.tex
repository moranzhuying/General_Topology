\section{商群}

记号约定:$G$是拓扑群(乘法记号),$H\unlhd G$.

\begin{proposition}{商群构成典范的拓扑群}{}
	$G/H$是拓扑群.
\end{proposition}

\begin{proposition}{商群中原点邻域基的性质}{}
	设$\phi:G\to G/H$是典范满射,$\mathfrak{B}$是$e$的邻域基,则$\phi\left(\mathfrak{B}\right)$是$\phi\left(e\right)$的邻域基.
\end{proposition}

\begin{proposition}{商群的基本拓扑性质}{}
	\begin{enumerate}
		\item $G/H$是Hausdorff的$\iff$ $H$是$G$的闭子群.
		\item $G/H$是离散的$\iff$ $H$是$G$的开子群.
		\item 如果$H$是离散的,那么$G/H$局部同构于$G$.
	\end{enumerate}
\end{proposition}

\begin{definition}{商群的Hausdorff化}{}
	我们称$G/\closure\left(\left\{e\right\}\right)$是$G$的Hausdorff化.
\end{definition}
