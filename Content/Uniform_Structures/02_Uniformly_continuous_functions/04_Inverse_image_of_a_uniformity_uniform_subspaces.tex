\section{一致结构的拉回,一致子空间}

\begin{definition}{拉回一致结构}{}
	设$X$是集合,$Y$是一致空间,配备一致结构$\mathfrak{U}$,$f\in\Hom_{\mathsf{Set}}\left(X,Y\right)$.称$\left(f\right)_{i\in\left\{1\right\}}$的始一致结构为$\mathfrak{U}$在$f$下的拉回(或$f$下的拉回一致结构).
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 记$g=f\times f$,则$\left\{g^{-1}\left(V\right)\middle|V\in\mathfrak{U}\right\}$是$\mathfrak{U}$在$f$的拉回下的邻里基.当$f$是满射时,$Y$中的每个邻里都是$X$中某个邻里在$g$下的像.
		\item $\mathfrak{U}$在$f$下的拉回诱导的拓扑等于$Y$的拓扑在$f$下的拉回.
	\end{enumerate}
\end{remark}

\begin{proposition}{一致连续性的拉回表征}{}
	设$X,Y$是一致空间,$f\in\Hom_{\mathsf{Set}}\left(X,Y\right)$,则$f\in\Hom_{\mathsf{Uni}}\left(X,Y\right)$ $\iff$ $Y$的一致结构在$f$下的拉回比$X$上的一致结构粗.
\end{proposition}

\begin{definition}{诱导一致结构,一致子空间}{}
	设$X$是一致空间,其一致结构为$\mathfrak{U}$,$A\subseteq X$,映射$i:A\to X$是典范嵌入.
	\begin{enumerate}
		\item 我们称$\mathfrak{U}$在$i$下的拉回是$\mathfrak{U}$在$A$上诱导的一致结构.
		\item 配备$\mathfrak{U}$在$A$上诱导的一致结构的$A$称为$X$的一致子空间.
	\end{enumerate}
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 记$\mathfrak{U}$在$A$上诱导的一致结构为$\mathfrak{U}_{A}$,则$\mathfrak{U}_{A}$在$A$上诱导的拓扑和拓扑空间$X$在$A$上诱导的子空间拓扑相同.
		\item 若$Y$是一致空间,$f\in\Hom_{\mathsf{Uni}}\left(X,Y\right)$,则$f|_{A}\in\Hom_{\mathsf{Uni}}\left(A,Y\right)$.
		\item 若$Y$是一致空间,$B\subseteq Y$,$f\in\Hom_{\mathsf{Uni}}\left(X,Y\right)$且$\im\left(f\right)\subseteq B$,则$\bar{f}:X\to B,x\mapsto f\left(x\right)$一致连续.
		\item 若$B\subseteq A\subseteq X$,则$X$在$B$上诱导的一致结构和$A$在$B$上诱导的一致结构相同.
	\end{enumerate}
\end{remark}

\begin{proposition}{稠密一致子空间的邻里基}{}
	设$X$是一致空间,$A$是$X$的稠密一致子空间,则$\left\{\closure\left(U\right)\middle|U\text{是}A\text{的邻里}\right\}$是$X$的邻里基.
\end{proposition}
