\chapter{指数和对数}

\section{以实数\texorpdfstring{$a$}{a}为底的指数和对数}

\begin{theorem}{}{}
	乘法群$\R_{>0}$和加法群$\R$作为拓扑群同构.
\end{theorem}

\begin{proposition}{实指数函数的存在性}{existence-and-uniqueness-of-real-expotential-function}
	对每个$a\in\R_{>0}$,存在唯一的$f_{a}\in\Hom_{\mathsf{TopGrp}}\left(\R,\R_{>0}\right)$使得$f_{a}\left(1\right)=a$.
\end{proposition}

\begin{definition}{以$a$为底的实指数函数和实对数函数}{}
	沿用\cref{existence-and-uniqueness-of-real-expotential-function}的记号.设$a\in\R_{>0}$.
	\begin{enumerate}
		\item 我们称$f_{a}$是以$a$为底的实指数函数.当$a\neq 1$时,记$\log_{a}\coloneq f_{a}^{-1}$,称$\log_{a}$是以$a$为底的实对数函数.
		\item 对每个$x\in\R$,记$a^{x}\coloneq f_{a}\left(x\right)$,称$a^{x}$是以$a$为底的指数.
		\item 当$a\neq 1$时,对每个$x\in\R_{>0}$,称$\log_{a}\left(x\right)$是以$a$为底的对数.
	\end{enumerate}
\end{definition}

\begin{proposition}{}{}
	对每个$f\in\Hom_{\mathsf{TopGrp}}\left(\R,\R_{>0}\right)$,存在$\lambda\in\R$使得$\forall x\in\R\left(f\left(x\right)=a^{\lambda x}\right)$.
\end{proposition}

\begin{corollary}{}{}
	\begin{enumerate}
		\item 对每个$x,y\in\R$和$a\in\R_{>0}$有$\left(a^{x}\right)^{y}=a^{xy}$.
		\item 对每个$a\in\R_{>0}\setminus\left\{1\right\},x\in\R_{>0}$和$y\in\R$有$x^{y}=a^{y\log_{a}\left(x\right)}$.
		\item 对每个$a\in\R_{>0}$和$n\geq 1$有$a^{\frac{1}{n}}=\sqrt[n]{a}$.
	\end{enumerate}
\end{corollary}

\begin{proposition}{}{}
	对每个$a,b\in\R_{>0}\setminus\left\{1\right\}$和$x\in\R_{>0}$有$\log_{a}\left(x\right)=\log_{a}\left(b\right)\log_{b}\left(x\right)$.
\end{proposition}

\begin{proposition}{}{}
	对每个$f\in\End_{\mathsf{TopGrp}}\left(\R_{>0}\right)$,存在唯一的$\alpha\in\R$使得$\forall x\in\R_{>0}\left(f\left(x\right)=x^{\alpha}\right)$.
\end{proposition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 由于对数函数是群同态,对数长期是数值计算的基本工具,用于此目的时取$a=10$,并有$\log_{10}\left(x\right)$的数值表.
		\item 在分析学中,我们经常采用另一个底$\mathrm{e}$,它满足
		\begin{align*}
			\limit_{\substack{x\to 1\\ x\in\R_{>0}\setminus\left\{1\right\}}}\frac{\log_{\mathrm{e}}\left(x\right)}{x-1}=1.
		\end{align*}
	\end{enumerate}
\end{remark}
