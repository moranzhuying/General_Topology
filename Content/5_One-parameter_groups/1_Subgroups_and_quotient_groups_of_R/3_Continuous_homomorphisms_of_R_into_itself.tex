\section{\texorpdfstring{$\R$}{R}上的拓扑群自同态}

\begin{proposition}{}{}
	设$f$是$\R$的加法群的自同态,则$\exists!a\in\R\left(f=a\id_{\R}\right)$.进一步,若$a\neq 0$,则$f$是拓扑群自同构.
\end{proposition}

\begin{corollary}{}{}
	$\R$上的所有拓扑群自同态构成的的加法群同构于$\R$.
\end{corollary}

\begin{corollary}{}{}
	$\R$上的所有拓扑群自同构按映射复合组成的乘法群同构于$\R^{\times}$.
\end{corollary}

\begin{proposition}{}{}
	设拓扑群$G$同构于加法群$\R$,则对任一$a\in G$,存在唯一的拓扑群同态$f_{a}:\R\to G$使得$f_{a}\left(1\right)=a$.
	
	进一步,若$a\neq 1_{G}$,则$f_{a}$是拓扑群同构.
\end{proposition}
