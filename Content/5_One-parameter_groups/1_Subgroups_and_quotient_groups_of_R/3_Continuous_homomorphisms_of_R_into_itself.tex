\section{\texorpdfstring{$\R$}{R}上的拓扑群自同态}

\begin{proposition}{}{}
	对加法群$\R$上的每个拓扑群自同态$f$,存在$a\in\R$使得$f$是$a$在$\R$上诱导的左乘.
\end{proposition}

\begin{corollary}{}{}
	加法群$\R$的自同构群同构于$\R^{\times}$.
\end{corollary}

\begin{corollary}{}{}
	设$G$是(作为拓扑群)同构于加法群$\R$的拓扑群(乘法记号).
	\begin{enumerate}
		\item 对每个$a\in G$,存在唯一的$f_{a}\in\Hom_{\mathsf{TopGrp}}\left(\R,G\right)$使得$f_{a}\left(1\right)=a$.
		\item 对每个$a\in G\setminus\left\{1\right\}$有$f_{a}\in\Isom_{\mathsf{TopGrp}}\left(\R,G\right)$.
	\end{enumerate}
\end{corollary}
