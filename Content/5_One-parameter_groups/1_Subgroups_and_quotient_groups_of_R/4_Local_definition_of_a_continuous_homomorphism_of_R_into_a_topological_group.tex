\section{从\texorpdfstring{$\R$}{R}到拓扑群的局部拓扑群同态的定义}

记号约定:$G$是拓扑群(乘法记号).

\begin{proposition}{}{}
	设$I$是$\R$中包含$0$的非退化区间,$f\in\Hom_{\mathsf{Top}}\left(I,G\right)$且
	\begin{align*}
		\forall x,y\in I\left(x+y\in I\implies f\left(x+y\right)=f\left(x\right)f\left(y\right)\right),
	\end{align*}
	那么存在唯一的$\tilde{f}\in\Hom_{\mathsf{TopGrp}}\left(\R,G\right)$使得$\tilde{f}|_{I}=f$.
\end{proposition}

\begin{corollary}{}{}
	设$f:\R\to G$是局部同构,则存在$G$的开子群$H$,$0$在$\R$中的邻域$U$和唯一的$\tilde{f}\in\Hom_{\mathsf{TopGrp}}$使$\tilde{f},f$在$U$上一致.
\end{corollary}

\begin{proposition}{}{}
	每个和加法群$\R$局部同构的连通拓扑群,要么同构于加法群$\R$,要么同构于$\T$.
\end{proposition}
