\section{\texorpdfstring{$\R$}{R}的商群}

\begin{proposition}{}{}
	设$H$是加法群$\R$的闭子群.如果$\R/H$非平凡,那么存在$a\geq 0$使得$\R/a\Z$.
\end{proposition}

\begin{proposition}{}{}
	设$a\in\R_{\geq 1}$.如果$\R/a\Z$非平凡且作为拓扑群不同构于$\R$,那么$\R/a\Z$作为拓扑群同构于$\R/\Z$.
\end{proposition}

\begin{definition}{模$a$实数加法群,一维环面}{}
	对每个$a\in\R_{>0}$,称拓扑群$\R/a\Z$为模$a$实数加法群.特别地,记$\T\coloneq\R/\Z$,称拓扑群$\T$为$1$维环面.
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 公式$x\equiv y\pmod{a\Z}$通常简写为$x\equiv y\pmod{a}$.当$a\in\Z$时,这和模$a$整数加法的记号一致.
		\item $\T$同胚于$\R^{2}$中的单位圆周,$\T^{2}$同胚于$\R^{3}$中的旋转环面.一般地,可以定义$n$维环面.
	\end{enumerate}
\end{remark}

\begin{proposition}{}{}
	\begin{enumerate}
		\item 对每个$\alpha\in\R$,定义$\left[\alpha,\alpha+1\right]$上的等价关系$\sim_{\alpha}$为
		\begin{align*}
			\forall x,y\in\left[\alpha,\alpha+1\right]\left(x\sim_{\alpha}y\iff\left(x=y\vee\left\{x,y\right\}\right)=\left\{\alpha,\alpha+1\right\}\right).
		\end{align*}
		那么对每个$\alpha\in\R$,$1$维环面同胚于$\left[\alpha,\alpha+1\right]/\sim_{\alpha}$.
		\item $\T$是紧,连通,局部连通的拓扑空间.
	\end{enumerate}
\end{proposition}

\begin{remark}{}{}
	设$\varphi:\R\to\T$是典范映射,$a\in\R$.
	\begin{enumerate}
		\item $\varphi|_{\left[a,a+1\right)}$是连续双射但不是同胚.
		\item 给$\left[a,a+1\right)$装备$\varphi|_{\left[a,a+1\right)}$诱导的拉回拓扑,该拓扑和$\R$诱导的子空间拓扑不同,并且使$\left[a,a+1\right)$同胚于$\T$
	\end{enumerate}
\end{remark}
