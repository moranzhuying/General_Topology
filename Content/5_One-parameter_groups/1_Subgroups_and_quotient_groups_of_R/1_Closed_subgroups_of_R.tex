\chapter{\texorpdfstring{$\R$}{R}的子群和商群}

\section{\texorpdfstring{$\R$}{R}的闭子群}

\begin{lemma}{}{}
	加法群$\R$的每个非离散子群都在$\R$中稠密.
\end{lemma}

\begin{proposition}{}{}
	如果$H$是加法群$\R$的非平凡闭子群,那么存在$a\in\R_{\geq 1}$使得$H=a\Z$.
\end{proposition}

\begin{corollary}{}{}
	对加法群$\R$的每个子群$H$,它要么在$\R$中稠密,要么是离散群.
\end{corollary}
