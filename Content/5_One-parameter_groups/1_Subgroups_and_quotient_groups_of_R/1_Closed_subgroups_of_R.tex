\chapter{\texorpdfstring{$\R$}{R}的子群和商群}

\section{\texorpdfstring{$\R$}{R}的闭子群}

\begin{lemma}{}{}
	$\R$的每一个$\R$不是离散群的加法子群都在$\R$中稠密.
\end{lemma}

\begin{proposition}{}{}
	设$H$是$\R$的加法闭子群,则如下三种情形中有且仅有一种成立:
	\begin{enumerate}
		\item $H=\R$;
		\item $H=\left\{0\right\}$;
		\item 存在唯一的$a\in\R_{>0}$使得$H=a\Z$.
	\end{enumerate}
	当(3)成立时,$H$是$G$的离散子群;当$H$不是$\R$的离散子群时,$H=\R$.
\end{proposition}
