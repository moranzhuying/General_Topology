\chapter{\texorpdfstring{$\R$}{R}和\texorpdfstring{$\T$}{T}的拓扑特征}

未加声明时,$G$是加法拓扑群.

\begin{theorem}{}{}
	若存在$1_{G}$的邻域$U$和$\R$的开区间$I$使得$U$和$I$同胚,则$G$局部同构于$\R$.
\end{theorem}

\begin{theorem}{}{}
	设$G$连通,并且存在$1_{G}$的邻域$U$和$\R$的开区间$I$使得$U$和$I$同胚.
	\begin{enumerate}
		\item 若$G$是紧\text{Hausdorff}群,则有拓扑群同构$G\simeq\T$.
		\item 若$G$不是紧群,则有拓扑群同构$G\simeq\R$.
	\end{enumerate}
\end{theorem}

\begin{corollary}{}{}
	若$G$作为拓扑空间和$\R$同胚,则$G$作为拓扑群和$\R$同构.
\end{corollary}
