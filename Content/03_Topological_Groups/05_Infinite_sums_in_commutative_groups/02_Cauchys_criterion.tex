\section{Cauchy收敛准则}

记号约定:$G$是交换拓扑群(加法记号).

\begin{theorem}{可和族的Cauchy准则}{}
	设$G$是Hausdorff的,$\left(x_{i}\right)_{i\in I}$是$G$中的元素族.
	\begin{enumerate}
		\item 若$\left(x_{i}\right)_{i\in I}$可和,则$\forall V\in\mathcal{N}\left(0\right)\exists J_{0}\in\fin\left(I\right)\forall J\in\fin\left(I\setminus J_{0}\right)\left(J\cap J_{0}=\emptyset\implies\sum\limits_{j\in J}x_{j}\in V\right)$.
		\item 设$G$完备.若$\forall V\in\mathcal{N}\left(0\right)\exists J_{0}\in\fin\left(I\right)\forall J\in\fin\left(I\setminus J_{0}\right)\left(J\cap J_{0}=\emptyset\implies\sum\limits_{j\in J}x_{j}\in V\right)$,则$\left(x_{i}\right)_{i\in I}$可和.
	\end{enumerate}
\end{theorem}

\begin{proposition}{Hausdorff拓扑群中的可和族按余有限滤子收敛到$0$}{}
	设$G$是Hausdorff的,$\left(x_{i}\right)_{i\in I}$是$G$的可和族,那么对每个$V\in\mathcal{N}\left(e\right)$,集合$\left\{i\in I\middle|x_{i}\notin V\right\}$有限.
\end{proposition}

\begin{corollary}{第一可数的拓扑群中的可和族一定有可数的支撑}{}
	设$G$的零元有一个可数邻域基.如果$\left(x_{i}\right)_{i\in I}$是$G$的可和族,那么$\left(x_{i}\right)_{i\in I}$的支撑是可数的.
\end{corollary}

\begin{corollary}{}{}
	设$G$是Hausdorff的完备拓扑群,它的单位元有一个由$G$的子群构成的邻域基,$\left(x_{i}\right)_{i\in I}$是$G$的元素族.那么
	\begin{center}
		$\left(x_{i}\right)_{i\in I}$可和$\iff$ $\left(x_{i}\right)_{i\in I}$按$I$的余有限滤子收敛到$0$.
	\end{center}
\end{corollary}
