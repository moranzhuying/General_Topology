\section{局部紧齐性空间}

记号约定:$G$是局部紧拓扑群.

\begin{proposition}{}{}
	若$H$是$G$的闭子群,则齐性空间$G/H$既是局部紧的仿紧空间.
\end{proposition}

\begin{proposition}{}{}
	$G$中所有开子集的交是$G$的单位分支.
\end{proposition}

\begin{corollary}{}{}
	设$G$完全不连通,那么$e$在$G$中的每个邻域都包含$G$的一个开子群.
\end{corollary}

\begin{corollary}{}{}
	如果$e$的每个邻域$V$都生成$G$,那么$G$是连通的.
\end{corollary}

\begin{corollary}{}{}
	设$H$是$G$的闭子群,$\varphi:G\to G/H$是典范映射,那么对$G$的每个连通分支$U$,集合$\closure
	\left(\varphi\left(U\right)\right)$是$G/H$的连通分支.
\end{corollary}
