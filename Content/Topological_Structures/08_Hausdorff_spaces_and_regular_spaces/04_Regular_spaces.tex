\section{正则空间}

记号约定:$X$是拓扑空间.

\begin{proposition}{正规性的等价刻画}{equivalent-condition-of-regularity}
	下列陈述等价:
	\begin{enumerate}
		\item ($\mathrm{O}_{3}$)$X$中每一点$x$的每个邻域都包含一个闭邻域.
		\item ($\mathrm{O}_{3}^{\prime}$)对$X$的任意闭集$F$和每个$x\in X\setminus F$,在$X$中$x$和$F$各自有一个邻域不相交.
	\end{enumerate}
\end{proposition}

\begin{definition}{正则空间}{}
	如果$X$是Hausdorff空间且满足命题(\ref{prop:equivalent-condition-of-regularity})中的陈述之一,则称$X$为正则空间,$X$配备的拓扑称为正则拓扑.
\end{definition}

\begin{example}{正则空间的例子}{}
	\begin{enumerate}
		\item 每个离散空间都是正则空间.
		\item 后面会看到,每个局部紧空间都是正则空间.特别地,实直线$\R$是正则空间.
	\end{enumerate}
\end{example}

\begin{proposition}{正则性可遗传到子空间}{}
	若$X$是正则空间,则$X$的每个子空间都是正则空间.
\end{proposition}

\begin{proposition}{局部闭邻域判断正则性}{}
	若$X$中每一点都有一个正则的闭邻域,则$X$是正则空间.
\end{proposition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 存在不是Hausdorff空间,但是每一点都有一个正则邻域的拓扑空间.
		\item 存在是Hausdorf空间,但不是正则空间的拓扑空间.
		\item 比正则拓扑细的拓扑不一定是正则拓扑.
	\end{enumerate}
\end{remark}
