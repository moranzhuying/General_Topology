\section{局部闭子空间}

记号约定:$X$是拓扑空间.

\begin{definition}{局部闭}{}
	设$L\subseteq X$.
	\begin{enumerate}
		\item 设$x\in L$.若$x$在$X$中有一个邻域$V$使得$L\cap V$是$V$的闭集,则称$L$在$x$处是局部闭的.
		\item 若对每个$x\in L$,集合$L$在$x$处都是局部闭的,则称$L$是$X$中的局部闭集.
	\end{enumerate}
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 设$F\subseteq X$.如果对每个$x\in X$,都存在$x$在$X$中的邻域$V$使得$F\cap V$是$V$的闭集,则$F$是$X$的闭集.
		\item 一般来说,$X$中有局部闭但不是闭集的集合.
	\end{enumerate}
\end{remark}

\begin{proposition}{局部闭集的等价刻画}{}
	设$L\subseteq X$.下列陈述等价:
	\begin{enumerate}
		\item $L$是$X$的局部闭集.
		\item 存在$X$中的开集$U$和闭集$V$使得$L=U\cap V$.
		\item $L$是$\closure_{X}\left(L\right)$的开集.
	\end{enumerate}
\end{proposition}

\begin{corollary}{连续映射保持局部闭集}{}
	设$Y$是拓扑空间,$L$是$Y$的局部闭集,$f:X\to Y$连续,则$f^{-1}\left(L\right)$是$X$的局部闭集.
\end{corollary}
