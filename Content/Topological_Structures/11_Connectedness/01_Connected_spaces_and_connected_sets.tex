\chapter{连通性}

\section{连通空间和连通集}

记号约定:$X$是拓扑空间.

\begin{definition}{连通空间,不连通空间}{}
	如果$X$不能表示成两个不相交非空开集的并,则称$X$是连通空间,否则称$X$是不连通空间.
\end{definition}

\begin{remark}{}{}
	下面是连通空间的另外几个等价定义:
	\begin{itemize}
		\item $X$不能表示成两个不相交非空闭集的并.
		\item $X$中的开闭集只有$\emptyset$和$X$.
	\end{itemize}
\end{remark}

\begin{remark}{}{}
	如果$X$中有一个由非空开集族构成的划分$\left(U_{i}\right)_{i\in I}$,那么$\left(U_{i}\right)_{i\in I}$是一族开闭集,并且$X$同胚于$\coprod\limits_{i\in I}U_{i}$.特别地,当$I$的基数$\geq 2$时,$X$是不连通空间.
\end{remark}

\begin{example}{连通空间的例子}{}
	\begin{enumerate}
		\item 实直线$\R$是连通空间,但有理直线$\Q$不是连通空间.
		\item 多于一个点的离散空间不是连通空间.
	\end{enumerate}
\end{example}

\begin{definition}{连通集,不连通集}{}
	设$A\subseteq X$.若$A$是连通空间(相对的,不连通空间),则称$A$是$X$的连通集(相对的,不连通集).
\end{definition}

\begin{remark}{}{}
	下面是连通集的另外几个等价定义:
	\begin{itemize}
		\item 对$X$的任意开集$U$和$V$,有$A\subseteq U\cap V\wedge A\cap U\neq\emptyset\wedge A\cap V\neq\emptyset\implies A\cap U\cap V=\emptyset$.
		\item 对$X$的任意闭集$U$和$V$,有$A\subseteq U\cap V\wedge A\cap U\neq\emptyset\wedge A\cap V\neq\emptyset\implies A\cap U\cap V=\emptyset$.
	\end{itemize}
\end{remark}

\begin{example}{连通集的例子}{}
	\begin{enumerate}
		\item 在每个拓扑空间中,空集和单点集都是连通集.
		\item 在Hausdorff空间中,每个基数$\geq 2$的有限集都是不连通的.更一般的,Hausdorff空间中每个基数$\geq 2$且至少有一个孤立点的子集都不连通.
	\end{enumerate}
\end{example}

\begin{remark}{}{}
	如果$A$是$X$的稠密子集,则$X$是连通空间.
\end{remark}

\begin{proposition}{闭包夹逼下的连通性遗传}{}
	设$A$是$X$的连通集.若$X$的子集$B$满足$A\subseteq B\subseteq\closure\left(A\right)$,则$B$是$A$的连通集.
\end{proposition}

\begin{proposition}{有交连通集的并仍然连通}{}
	设$\left(A_{i}\right)_{i\in I}$是$X$的一族连通集.若$\bigcap\limits_{i\in I}A_{i}\neq\emptyset$,则$\bigcup\limits_{i\in I}A_{i}$是$X$的连通集.
\end{proposition}

\begin{corollary}{有交链式连通集的并仍然连通}{}
	设$\left(A_{n}\right)_{n=0}^{\infty}$是$X$中的连通集序列.若$\forall n\in\N\left(A_{n+1}\cap A_{n}\neq\emptyset\right)$,则$\bigcup\limits_{n=0}^{\infty}A_{n}$是$X$的连通集.
\end{corollary}

\begin{proposition}{连通集穿过边界}{}
	设$A\subseteq X$,$B$是$X$的连通集.若$B\cap A\neq\emptyset$且$B\cap\left(X\setminus A\right)\neq\emptyset$,则$B\cap\boundary\left(A\right)\neq\emptyset$.
\end{proposition}

\begin{corollary}{连通空间的非空真子集有边界}{}
	设$X$是连通空间.若$A$是$X$的非空真子集,则$\boundary\left(A\right)\neq\emptyset$.
\end{corollary}
