\section{紧空间的正则性}

记号约定:$X$是紧空间.

\begin{proposition}{紧空间中邻域基的判定}{}
	设$a\in X$,它在$X$中的闭邻域构成的滤子基为$\mathfrak{B}$,则$\mathfrak{B}$是$a$的邻域基$\iff$ $\bigcap\limits_{B\in\mathfrak{B}}B=\left\{x\right\}$.
\end{proposition}

\begin{corollary}{紧空间的正则性}{}
	每个紧空间都是正则空间.
\end{corollary}

\begin{proposition}{紧空间中不相交的闭集可以被开集分离}{}
	设$A$和$B$是$X$中两个不相交的闭集,则$X$中存在两个不相交的开集$U$和$V$,二者分别包含$A$和$B$.
\end{proposition}
