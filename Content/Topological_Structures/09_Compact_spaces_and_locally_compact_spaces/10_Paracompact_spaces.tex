\section{仿紧空间}

记号约定:$X$是拓扑空间.

\begin{definition}{仿紧空间}{}
	设$X$是Hausdorff空间.若$X$满足如下条件($\mathrm{PC}$):``$X$的每个开覆盖都有一个局部有限的开加细'',则称$X$是仿紧空间.
\end{definition}

\begin{remark}{}{}
	每个紧空间都是仿紧空间.每个离散空间都是仿紧空间.
\end{remark}

\begin{proposition}{闭子空间的仿紧性}{}
	若$X$是仿紧空间,则$X$的每个闭子空间都是仿紧空间.
\end{proposition}

\begin{remark}{}{}
	紧空间的开子空间不一定是仿紧空间.
\end{remark}

\begin{proposition}{仿紧空间与紧空间的乘积仍然仿紧}{}
	若$X$是仿紧空间,$Y$是紧空间,则$X\times Y$是仿紧空间.
\end{proposition}

\begin{remark}{}{}
	两个仿紧空间的乘积不一定是仿紧空间.
\end{remark}

\begin{proposition}{仿紧空间族的无交并还是仿紧空间}{}
	若$\left(X_{i}\right)_{i\in I}$是仿紧空间族,则$\coprod\limits_{i\in I}X_{i}$是仿紧空间.
\end{proposition}

\begin{proposition}{局部紧空间的仿紧性判定}{}
	设$X$是局部紧空间,则$X$是仿紧空间$\iff$ $X$同胚于一族$\sigma$-紧空间的无交并.
\end{proposition}

\begin{corollary}{仿紧局部紧空间的开覆盖加细}{}
	设$X$是局部紧的仿紧空间.
	\begin{enumerate}
		\item $X$的每个开覆盖都有一个由相对紧集组成的局部有限开加细.
		\item 若$X$是$\sigma$-紧空间,则(1)中的局部有限开加细可以是可数的.
	\end{enumerate}
\end{corollary}
