\section{\texorpdfstring{$\sigma$}{sigma}-紧空间}

记号约定:$X$是局部紧空间.

\begin{definition}{$\sigma$-紧空间}{}
	若$X$可以表示为$X$中可数个紧子集的并,则称$X$是$\sigma$-紧空间.
\end{definition}

\begin{example}{$\sigma$-紧空间的例子}{}
	\begin{enumerate}
		\item 一个离散空间是$\sigma$-紧空间$\iff$这个离散空间是可数集.
		\item 实直线$\R$是$\sigma$-紧空间,因为它是$\left(\left[-n,n\right]\right)_{n=0}^{\infty}$的并.
	\end{enumerate}
\end{example}

\begin{remark}{}{}
	一个Hausdorff空间可以是可数个紧子空间的并,但并不是局部紧空间,例如配备弱拓扑的Hillbert空间.
\end{remark}

\begin{proposition}{$\sigma$-紧空间中紧集套的存在性}{existence-of-compact-set-sequence}
	若$X$是$\sigma$-紧空间,则$X$中有一族相对紧的开集,满足$\forall n\in\N\left(\closure\left(U_{n}\right)\subseteq U_{n+1}\right)$.
\end{proposition}

\begin{corollary}{}{}
	设$X$是$\sigma$-紧空间,$\left(U_{n}\right)_{n=0}^{\infty}$如命题(\ref{prop:existence-of-compact-set-sequence})所定义,则$X$中每个紧集都被$\left(U_{n}\right)_{n=0}^{\infty}$的某项包含.
\end{corollary}

\begin{remark}{}{}
	\begin{enumerate}
		\item 设$X$的单点紧化是$X^{\prime}$,无穷远点为$\omega$,则$X$是$\sigma$-紧空间$\iff$ $\omega$在$X$中有一个可数邻域基.
		\item 若$X$是$\sigma$-紧空间,则$X$的每个闭子空间都是$\sigma$-紧空间.
		\item 有限个$\sigma$-紧空间的积空间还是$\sigma$-紧空间.
	\end{enumerate}
\end{remark}

\begin{remark}{}{}
	紧空间的开子空间不一定是$\sigma$-紧空间.
\end{remark}
