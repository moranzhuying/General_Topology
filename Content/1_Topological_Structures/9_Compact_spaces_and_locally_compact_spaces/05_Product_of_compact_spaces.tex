\section{紧空间的乘积}

记号约定:$\left(X_{i}\right)_{i\in I}$是拓扑空间族,$X=\prod\limits_{i\in I}X_{i}$.

\begin{theorem}{Tychonoff定理}{}
	\begin{enumerate}
		\item 若$\left(X_{i}\right)_{i\in I}$是拟紧空间族(相对的,紧空间族),则$X$是拟紧空间(相对的,紧空间).
		\item 若$\left(X_{i}\right)_{i\in I}$是一族非空拓扑空间,$X$是拟紧空间(相对的,紧空间),则$\left(X_{i}\right)_{i\in I}$是拟紧空间族(相对的,紧空间族).
	\end{enumerate}
\end{theorem}

\begin{corollary}{相对拟紧集的积空间判定}{}
	设$A\subseteq X$,则$A$是$X$的相对紧集$\iff$对每个$i\in I$,集合$\pr_{i}\left(A\right)$是$X_{i}$的相对紧集.
\end{corollary}
