\section{位移}

记号约定:$d$是$\R^{n}$上的Euclid度量,$\left\|\cdot\right\|$是$\R^{n}$上的Euclid范数.

\begin{definition}{Euclid位移}{}
	设$f\in\End_{\mathsf{Aff}}\left(\R^{n}\right)$.如果$d\circ\left(f\times f\right)=d$,则称$f$是Euclid位移.
\end{definition}

\begin{definition}{位移群,正交变换群}{}
	我们称$\Disp\left(\R^{n}\right)$是$\R^{n}$的Euclid位移群,称$\mathrm{O}_{n}\left(\R\right)$是$n$元实正交群,其元素称为实正交变换.
\end{definition}

\begin{definition}{$\R^{n}$上的标准内积}{}
	我们把和$\frac{1}{2}\left\|\cdot\right\|^{2}$诱导的对称双线性形式记作$\langle\cdot\mid\cdot\rangle$,称之为$\R^{n}$上的标准内积.
\end{definition}

\begin{proposition}{正交变换的刻画}{}
	设$\langle\cdot\mid\cdot\rangle$是$\R^{n}$上的标准内积,$f\in\End_{\R}\left(\R^{n}\right)$.下列陈述等价:
	\begin{enumerate}
		\item $f\in\mathrm{O}_{n}\left(\R\right)$.
		\item $\forall\mathbf{x}\in\R^{n}\left(\left\|f\left(\mathbf{x}\right)\right\|=\left\|\mathbf{x}\right\|\right)$.
		\item $\forall\mathbf{x}\in\R^{n}\left(\left\|f\left(\mathbf{x}\right)\right\|^{2}=\left\|\mathbf{x}\right\|^{2}\right)$.
		\item $\forall\mathbf{x},\mathbf{y}\in\R^{n}\left(\langle f\left(\mathbf{x}\right)\mid f\left(\mathbf{y}\right)\rangle=\langle\mathbf{x}\mid\mathbf{y}\rangle\right)$.
	\end{enumerate}
\end{proposition}
