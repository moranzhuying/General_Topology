\section{向量空间\texorpdfstring{$\R^{n}$}{Rn}}

记号约定:$m\geq 0$.

\begin{proposition}{$\R^{n}$是拓扑$\R$-模}{}
	$\R$在$\R^{n}$上的标量乘法$\R\times\R^{n}\to\R^{n},\left(t,\left(x_{i}\right)_{i=1}^{n}\right)\mapsto\left(tx_{i}\right)_{i=1}^{n}$是连续的.
\end{proposition}

\begin{proposition}{$\R^{n}$的维数}{}
	$\R^{n}$是$n$维$\R$-向量空间.
\end{proposition}

\begin{proposition}{}{}
	设$\left(\mathbf{e}_{i}\right)_{i=1}^{n},\left(\check{\mathbf{e}}_{j}\right)_{j=1}^{m}$分别是$\R^{n},\R^{m}$的典范基,$f\in\Hom_{\mathsf{Aff}}\left(\R^{n},\R^{m}\right),g:\R^{n}\to\R^{m},\mathbf{x}\mapsto f\left(\mathbf{x}\right)-f\left(\mathbf{0}\right)$.
	\begin{enumerate}
		\item $g\in\Hom_{\R}\left(\R^{n},\R^{m}\right)$.
		\item 设$\mathbf{x}=\sum\limits_{i=1}^{n}x_{i}\mathbf{e}_{i}\in\R^{n},f\left(\mathbf{0}\right)=\sum\limits_{k=1}^{m}b_{k}\check{e}_{k}$,对每个$1\leq i\leq n$有$g\left(\mathbf{e}_{i}\right)=\sum\limits_{j=1}^{m}a_{ij}\check{\mathbf{e}}_{j},f\left(\mathbf{e}_{i}\right)=\sum\limits_{j=1}^{m}y_{j}\check{\mathbf{e}}_{j}$,则
		\begin{align*}
			\forall 1\leq j\leq m\left(y_{j}=\sum\limits_{i=1}^{n}x_{i}a_{ij}+b_{j}\right).
		\end{align*}
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	每个$f\in\Hom_{\mathsf{Aff}}\left(\R^{n},\R^{m}\right)$都一致连续.
\end{proposition}

\begin{proposition}{}{}
	\begin{enumerate}
		\item 每个是满射的仿射映射$f:\R^{n}\to\R^{n}$都是双射,并且$f^{-1}$是仿射映射.
		\item 每个是满射的仿射映射$f:\R^{n}\to\R^{n}$都是同胚,并且是一致自同构.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	设$\left(\mathbf{a}_{i}\right)_{i=1}^{n}$是$\R^{n}$的线性无关族,$\mathbf{b}\in\R^{n}$.记$P\coloneq\mathbf{b}+\sum\limits_{i=1}^{n}\left[-1,1\right]\mathbf{a}_{i}$,其中$\left[-1,1\right]$是$\R$中的区间.
	\begin{enumerate}
		\item $P$是$\mathbf{b}$的紧邻域.
		\item $\interior\left(P\right)=\mathbf{b}+\sum\limits_{i=1}^{n}\left(-1,1\right)\mathbf{a}_{i}$.
	\end{enumerate}
\end{proposition}

\begin{definition}{闭平行多面体,开平行多面体}{}
	设$\left(\mathbf{a}_{i}\right)_{i=1}^{n}$是$\R^{n}$的线性无关族,$\mathbf{b}\in\R^{n}$.
	\begin{enumerate}
		\item 我们称$\mathbf{b}+\sum\limits_{i=1}^{n}\left[-1,1\right]\mathbf{a}_{i}$是以$\mathbf{b}$为中心,由$\left(\mathbf{a}_{i}\right)_{i=1}^{n}$张成的闭平行多面体.
		\item 我们称$\mathbf{b}+\sum\limits_{i=1}^{n}\left(-1,1\right)\mathbf{a}_{i}$是以$\mathbf{b}$为中心,由$\left(\mathbf{a}_{i}\right)_{i=1}^{n}$张成的开平行多面体.
	\end{enumerate}
\end{definition}

\begin{proposition}{}{}
	对各$f\in\R[X_{1},\ldots,X_{n}]$,记$\tilde{f}:\R^{n}\to\R$是$f$诱导的多项式函数,则$\R[X_{1},\ldots,X_{n}]\to\Pol\left(\R^{n},\R\right),P\mapsto\tilde{P}$是双射.
\end{proposition}

\begin{proposition}{}{}
	设$f\in\Pol\left(\R^{n},\R\right)$,那么$f$连续.进一步,若$f\neq 0$,则$\R^{n}\setminus f^{-1}\left(\left\{\mathbf{0}\right\}\right)$是$\R^{n}$的稠密开集.
\end{proposition}
