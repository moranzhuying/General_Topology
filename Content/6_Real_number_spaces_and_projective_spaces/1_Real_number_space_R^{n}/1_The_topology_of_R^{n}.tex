\chapter{实数空间\texorpdfstring{$\R^{n}$}{Rn}}

\section{\texorpdfstring{$\R^{n}$}{Rn}的拓扑}

\begin{definition}{}{}
	给$\R^{n}$配备$n$个拓扑群$\R$的乘积拓扑.我们称拓扑空间$\R^{n}$是$n$维实数空间.
\end{definition}

\begin{proposition}{}{}
	$\left|\R^{n}\right|=\left|\R\right|$.
\end{proposition}

\begin{definition}{}{}
	设$U\subseteq\R^{n}$.
	\begin{itemize}
		\item 若存在$\R$中的开区间族$\left(I_{k}\right)_{k=1}^{n}$使得$U=\prod\limits_{k=1}^{n}I_{k}$,则称$U$是$\R^{n}$中的开箱.进一步,若$\left(I_{k}\right)_{k=1}^{n}$是一族长度相等的有界开区间,则称$U$是$\R^{n}$中的开长方体,称$\ell\left(I_{1}\right)$为边长.
		\item 若存在$\R$中的闭区间族$\left(I_{k}\right)_{k=1}^{n}$使得$U=\prod\limits_{k=1}^{n}I_{k}$,则称$U$是$\R^{n}$中的闭箱.进一步,若$\left(I_{k}\right)_{k=1}^{n}$是一族长度相等的有界闭区间,则称$U$是$\R^{n}$中的闭长方体,称$\ell\left(I_{1}\right)$为边长.
	\end{itemize}
\end{definition}

\begin{definition}{}{}
	$\R^{2}$中的开箱,闭箱,开正方体,闭正方体依次称为$\R^{2}$中的开矩形,闭矩形,开正方形,闭正方形.
\end{definition}

\begin{proposition}{}{}
	记$\mathcal{B}=\left\{U\subseteq\R^{n}\middle|U\text{是开正方体}\right\}$,则$\mathcal{B}$是$\R^{n}$的拓扑基.
\end{proposition}

\begin{proposition}{}{}
	设$x\in\R^{n}$,记
	\begin{align*}
		\mathcal{B}_{1}=\left\{U\subseteq\R^{n}\middle|x\in U,U\text{是开正方体}\right\},\mathcal{B}_{2}=\left\{U\subseteq\R^{n}\middle|x\in\interior\left(U\right),U\text{是闭正方体}\right\},
	\end{align*}
	则$\mathcal{B}_{1}$和$\mathcal{B}_{2}$都是$x$的邻域基.
\end{proposition}

\begin{corollary}{}{}
	设$x=\left(x_{1},\ldots,x_{n}\right)\in\R^{n}$.对任一$m\in\Z_{\geqslant 1}$,记$K_{m}=\prod\limits_{k=1}^{n}\left(x_{i}-\frac{1}{k},x_{i}+\frac{1}{k}\right)$,则
	\begin{align*}
		\left\{K_{m}\subseteq\R^{n}\middle|m\in\Z_{\geqslant 1}\right\}
	\end{align*}
	是$x$的可数邻域基.
\end{corollary}

\begin{proposition}{}{}
	$\R^{n}$中的每个非空开立方体都和$\R^{n}$同胚.进一步,$\R^{n}$的每个非空开集$U$的基数等于$\left|\R\right|$.
\end{proposition}

\begin{proposition}{}{}
	$\R^{n}$的每个开立方体和闭立方体都是连通集.特别地,$\R^{n}$是连通且局部连通的拓扑空间.
\end{proposition}

\begin{corollary}{}{}
	设$A$是$\R^{n}$的非空开集,则$A$在$\R^{n}$中的每个连通分支都是开集,并且$A$的连通分支组成的集合可数.
\end{corollary}

\begin{definition}{}{}
	设$A\subseteq\R^{n}$.若对任一$i\in\llbracket 1,n\rrbracket$,$\pr_{i}\left(A\right)$在$\R$中按$\leqslant$有界,则称$A$在$\R^{n}$中有界.
\end{definition}

\begin{proposition}{}{}
	设$A\subseteq\R^{n}$,则$A$是$\R^{n}$的相对紧集$\iff$ $A$在$\R^{n}$中有界.
\end{proposition}

\begin{corollary}{}{}
	$\R^{n}$是局部紧空间,当$n\geqslant 2$时,$\R^{n}$不是紧空间.
\end{corollary}
