\chapter{实数空间\texorpdfstring{$\R^{n}$}{Rn}}

记号约定:$n\geq 0$.

\section{\texorpdfstring{$\R^{n}$}{Rn}的拓扑}

\begin{definition}{$n$维实数空间}{}
	拓扑空间族$\left(\R\right)_{k=1}^{n}$的乘积空间$\R^{n}$称为$n$维实数空间.
\end{definition}

\begin{remark}{}{}
	根据集合论的结果,当$n\geq 1$时$\left|\R^{n}\right|=\left|\R\right|$.
\end{remark}

\begin{definition}{开箱,闭箱}{}
	\begin{enumerate}
		\item $\R$中$n$个开区间的乘积称为$\R^{n}$中的开箱,$\R$中$n$个闭区间的乘积称为$\R^{n}$中的闭箱.
		\item $\R^{2}$中的开箱称为$\R$中的开矩形,$\R^{2}$中的闭箱称为$\R^{2}$中的闭矩形.
	\end{enumerate}
\end{definition}

\begin{proposition}{开箱和闭箱的拓扑性质}{}
	\begin{enumerate}
		\item $\R^{n}$中的开箱构成的集合是$\R^{n}$的乘积拓扑的基.
		\item $\R^{n}$中的每个闭箱都是$\R^{n}$的闭集.
		\item $\R^{n}$中每个开箱的闭包都是$\R^{n}$中的闭箱,$\R^{n}$中每个闭箱的内部都是$\R^{n}$的开箱.
		\item 设$\mathbf{x}\in\R^{n}$,那么$\R^{n}$中所有包含$\mathbf{x}$的开箱构成的集合是$\mathbf{x}$的邻域基,$\R^{n}$中所有以$\mathbf{x}$为内点的闭箱构成的集合是$\mathbf{x}$的邻域基.
		\item $\R^{n}$中的每个非空开箱都与$\R^{n}$同胚.
		\item 当$n\geq 1$时,$\R^{n}$中的每个非空开集都和$\R$等势.
	\end{enumerate}
\end{proposition}

\begin{definition}{开立方体,闭立方体,开(闭)立方体的边长}{}
	\begin{enumerate}
		\item $\R$中$n$个长度相等的有界开区间的乘积称为$\R^{n}$的开立方体,$\R$中$n$个长度相等的有界闭区间的乘积称为$\R^{n}$的闭立方体.
		\item $\R^{2}$中的开立方体称为$\R^{2}$中的开正方形,$\R^{2}$中的闭立方体称为$\R^{2}$中的闭正方形.
		\item 设$U$是$\R^{n}$中的开立方体(或闭立方体).构成$U$的每一个区间的长度称为$U$的边长.
	\end{enumerate}
\end{definition}

\begin{proposition}{$\R^{n}$中各点都有可数邻域基}{}
	设$\mathbf{x}\coloneq\left(x_{i}\right)_{i=1}^{n}\in\R^{n}$,$\left(k_{i}\right)_{i=0}^{\infty}$是$\N$中递增且收敛到$+\infty$的序列.对每个$1\leq i\leq m$,记
	\begin{align*}
		K_{m}=\prod\limits_{i=1}^{m}\left(x_{i}-\frac{1}{k_{i}},x_{i}+\frac{1}{k_{i}}\right),L_{i}=\prod\limits_{j=1}^{m}\left(x_{i}-\frac{1}{j},x_{i}+\frac{1}{j}\right),
	\end{align*}
	那么$\left\{K_{i}\middle|i\geq 1\right\}$和$\left\{L_{i}\middle|i\geq 1\right\}$是$\mathbf{x}$的可数邻域基.
\end{proposition}

\begin{proposition}{$\R^{n}$的开箱和闭箱的连通性}{}
	$\R^{n}$中的每个开箱和闭箱都是连通的.
\end{proposition}

\begin{corollary}{$\R^{n}$的连通性和局部连通性}{}
	$\R^{n}$连通且局部连通.
\end{corollary}

\begin{remark}{}{}
	\begin{enumerate}
		\item $\Q^{n}$是$\R^{n}$的一个可数稠密子集,因此$\R^{n}$可分.
		\item 如果$A$是$\R^{n}$的非空开集,那么$A$的每个连通分支都是开集,并且$A$的所有连通分支构成的集合可数.
	\end{enumerate}
\end{remark}

\begin{definition}{$\R^{n}$中的有界集}{}
	设$A\subseteq\R^{n}$.如果$\left(\pr_{i}\left(A\right)\right)_{i=1}^{n}$是$\R$的一族有界集,则称$A$在$\R^{n}$中有界.
\end{definition}

\begin{proposition}{相对紧$\iff$有界}{}
	设$A\subseteq\R^{n}$,则$A$在$\R^{n}$中相对紧$\iff$ $A$在$\R^{n}$中有界.
\end{proposition}

\begin{corollary}{$\R^{n}$的局部紧性和紧性}{}
	$\R^{n}$是局部紧空间.当$n\geq 1$时,$\R^{n}$不是紧空间.
\end{corollary}
