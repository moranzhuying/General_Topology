\section{域\texorpdfstring{$\R$}{R}上的向量空间与代数的拓扑}

\begin{proposition}{有限维实向量空间上典范拓扑的构造}{construction-of-canonical-topology-over-finite-dimensional-real-vector-space}
	设$E$是$n$维$\R$-向量空间,$\mathbf{v}\coloneq\left(\mathbf{v}_{i}\right)_{i=1}^{n}$是$V$的基,$\varphi_{\mathbf{v}}:\R^{n}\to E,\left(x_{i}\right)_{i=1}^{n}\mapsto\sum\limits_{i=1}^{n}x_{i}\mathbf{v}_{i}$.令$\tau$是$\R^{n}$上的拓扑在$\varphi_{\mathbf{v}}$下的终拓扑.那么:
	\begin{enumerate}
		\item 装备$\tau$的$E$是拓扑$\R$-模.
		\item $\tau$是$E$上唯一一个让$E$是拓扑$\R$-模的Hausdorff拓扑.
	\end{enumerate}
\end{proposition}

\begin{definition}{实向量空间上的典范拓扑}{}
	沿用\cref{prop:construction-of-canonical-topology-over-finite-dimensional-real-vector-space}的记号和设定.我们称$\tau$是$E$上的典范拓扑.
\end{definition}

\begin{proposition}{装备典范拓扑的有限维$\R$-代数是拓扑环}{}
	设$A$是$n$维$\R$-代数.把$A$视为$\R$-向量空间并为其装备典范拓扑,那么$A$是拓扑环.
\end{proposition}

\begin{proposition}{有限维$\R$-向量空间之间的双线性映射一定连续}{}
	设$E,F,G$是有限维$\R$-向量空间,则每个$B\in\Bil_{\R}\left(E,F;G\right)$都连续.
\end{proposition}
