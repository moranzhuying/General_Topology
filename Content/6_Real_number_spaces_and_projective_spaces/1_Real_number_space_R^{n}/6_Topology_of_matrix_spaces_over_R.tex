\section{\texorpdfstring{$\R$}{R}上矩阵空间的拓扑}

本小节中,默认$m,p\in\Z_{\geqslant 1}$.

\begin{definition}{}{}
	设$\left(\boldsymbol{E}_{i,j}\right)_{\substack{1\leqslant i\leqslant m\\1\leqslant j\leqslant n}}$是$\Mat_{m,n}\left(\R\right)$的标准基.记
	\begin{align*}
		\varphi:\R^{mn}\to\Mat_{m,n}\left(\R\right),\left(a_{1,1},\ldots,a_{m,n}\right)\mapsto\sum\limits_{\left(i,j\right)\in\llbracket 1,m\rrbracket\times\llbracket 1,n\rrbracket}a_{i,j}\boldsymbol{E}_{i,j}.
	\end{align*}
	给$\Mat_{m,n}\left(\R\right)$配备由$\varphi$和$\R^{mn}$诱导的典范拓扑.
\end{definition}

\begin{proposition}{}{}
	矩阵乘法$\cdot:\Mat_{m,n}\left(\R\right)\times\Mat_{n,p}\left(\R\right)\to\Mat_{m,p}\left(\R\right)$是连续映射.
\end{proposition}

\begin{corollary}{}{}
	$\Mat_{n}\left(\R\right)$按其上的典范拓扑构成拓扑幺环.
\end{corollary}

\begin{proposition}{}{}
	$\GL_{n}\left(\R\right)$是$\Mat_{n}\left(\R\right)$的稠密开集.
\end{proposition}
