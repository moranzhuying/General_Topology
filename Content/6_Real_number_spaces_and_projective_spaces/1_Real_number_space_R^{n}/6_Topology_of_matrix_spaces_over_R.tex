\section{\texorpdfstring{$\R$}{R}上矩阵空间的拓扑}

记号约定:$m,n,p\geq 0$.

\begin{definition}{矩阵空间}{}
	给$\Mat_{m,n}\left(\R\right)$装备典范拓扑,得到的拓扑$\R$-模称为$mn$维$\R$-矩阵空间.
\end{definition}

\begin{proposition}{矩阵乘法连续}{}
	$\Mat_{m,n}\left(\R\right)\times\Mat_{n,p}\left(\R\right)\to\Mat_{m,p}\left(\R\right),\left(\bm{X},\bm{Y}\right)\mapsto\bm{X}\bm{Y}$是连续映射.
\end{proposition}

\begin{corollary}{矩阵环是拓扑环}{}
	$\Mat_{n}\left(\R\right)$在典范拓扑下构成拓扑环.
\end{corollary}

\begin{proposition}{$\GL_{n}\left(\R\right)$是拓扑群}{}
	$\GL_{n}\left(\R\right)$是$\Mat_{n}\left(\R\right)$的稠密开集,它在装备了$\Mat_{n}\left(\R\right)$诱导的子空间拓扑后是拓扑群.
\end{proposition}

\begin{proposition}{取逆映射连续}{}
	$\GL_{n}\left(\R\right)\to\GL_{n}\left(\R\right),\bm{X}\mapsto\bm{X}^{-1}$是连续映射.
\end{proposition}
