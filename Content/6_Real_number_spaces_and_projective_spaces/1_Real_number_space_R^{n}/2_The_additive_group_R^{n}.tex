\section{加法群\texorpdfstring{$\R^{n}$}{Rn}}

\begin{proposition}{$\R^{n}$是拓扑群}{}
	加法群$\R^{n}$在乘积拓扑下是拓扑群.
\end{proposition}

\begin{definition}{$n$维实数空间的拓扑群,$\R^{n}$的加法一致结构}{}
	拓扑群$\R^{n}$称为$n$维实数空间的加法群,其一致结构称为$\R^{n}$的加法一致结构.
\end{definition}

\begin{remark}{}{}
	$\R^{n}$的加法一致结构是$\left(\R\right)_{k=1}^{n}$中各项的一致结构的乘积一致结构.对每个$p\geq 1$,记
	\begin{align*}
		V_{p}=\left\{\left(\left(x_{i}\right)_{i=1}^{n},\left(y_{i}\right)_{i=1}^{n}\right)\in\R^{n}\times\R^{n}\middle|\max\limits_{1\leq i\leq n}\left|x_{i}-y_{i}\right|<\frac{1}{p}\right\},
	\end{align*}
	那么$\left\{V_{p}\middle|p\geq 1\right\}$是$\R^{n}$的加法一致结构的邻里基.
\end{remark}

\begin{proposition}{$\R^{n}$的完备性}{}
	$\R^{n}$是完备一致空间.
\end{proposition}
