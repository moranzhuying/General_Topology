\section{加法群\texorpdfstring{$\R^{n}$}{Rn}}

\begin{definition}{}{}
	$\R^{n}$上的乘积拓扑使得加法群$\R^{n}$成为拓扑群.当$n=0$时,记$\R^{n}=\left\{0\right\}$.我们称加法拓扑群$\R^{n}$是$n$维实数空间的加法群.
\end{definition}

\begin{definition}{}{}
	$\R^{n}$的加法一致结构定义为$\R$的一致结构的乘积.对任一$p\in\Z_{\geqslant 1}$,记
	\begin{align*}
		V_{p}=\left\{\left(\left(x_{1},\ldots,x_{n}\right),\left(y_{1},\ldots,y_{n}\right)\right)\in\R^{n}\times\R^{n}\middle|\max_{1\leqslant i\leqslant n}\left|x_{i}-y_{i}\right|\leqslant\frac{1}{p}\right\},
	\end{align*}
	则$\left\{V_{p}\subseteq\R^{n}\times\R^{n}\middle|p\in\Z_{\geqslant 1}\right\}$是$\R^{n}$的一致结构的基.论及$\R^{n}$的一致结构,在没有声明时总是默认指其加法一致结构.
\end{definition}

\begin{proposition}{}{}
	$\R^{n}$按照其加法一致结构成为完备空间.
\end{proposition}
