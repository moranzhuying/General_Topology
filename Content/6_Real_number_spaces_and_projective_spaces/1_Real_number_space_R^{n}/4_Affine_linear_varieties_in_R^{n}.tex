\section{\texorpdfstring{$\R^{n}$}{Rn}中的仿射子空间}

本节中,默认$p\in\Z_{\geqslant 1}$,$\left(e_{i}\right)_{i=1}^{n}$是$\R^{n}$的标准基.

\begin{definition}
	对任一$i\in\llbracket 1,n\rrbracket$,直线$\R e_{i}$称为$\R^{n}$中的第$i$条坐标轴.
\end{definition}

\begin{definition}
	设$n=2$时,$x=\left(x_{1},x_{2}\right)\in\R^{2}$.我们称$\R e_{1}$是横轴,$\R e_{2}$是纵轴,$x_{1}$,$x_{2}$依次称为$x$的横坐标和纵坐标.
\end{definition}

\begin{proposition}
	$\R^{n}$中的每个$p$维仿射线性簇都是$\R^{n}$的闭集,并且与$\R^{p}$同胚.
\end{proposition}

\begin{proposition}
	设$a\in\R^{n}$,$D$是$\R^{n}$中过$a$的直线,则存在$b\in\R^{n}\setminus\left\{0\right\}$使得$D=a+\R b$.此外,
	\begin{align*}
		\varphi:\R\to D,t\mapsto a+tb
	\end{align*}
	是同胚.
\end{proposition}

\begin{definition}
	设$a\in\R^{n}$,$D$是$\R^{n}$中过$a$的直线,$b\in\R^{n}\setminus\left\{0\right\}$且$D=a+\R b$.
	\begin{itemize}
		\item 我们称$b$是$D$的一个方向向量.
		\item 设$b=\left(b_{1},\ldots,b_{n}\right)$.我们称$\left(b_{1},\ldots,b_{n}\right)$是$D$的方向比.
	\end{itemize}
\end{definition}

\begin{proposition}
	设$a\in\R^{n}$,$D$是$\R^{n}$中过$a$的直线,$b,b^{\prime}\in\R^{n}\setminus\left\{0\right\}$是$D$的方向向量,则$\exists\lambda\in\R\setminus\left\{0\right\}\left(b=\lambda b^{\prime}\right)$.
\end{proposition}

\begin{definition}
	设$a\in\R^{n}$,$b\in\R^{n}\setminus\left\{0\right\}$.
	\begin{itemize}
		\item $a+\R_{\geqslant 0}b$称为以$a$为原点,$b$为方向向量的闭射线(射线,班直线).
		\item $a+\R_{>0}b$称为以$a$为原点,$b$为方向向量的开射线.
	\end{itemize}
	可以验证:
\end{definition}

\begin{proposition}
	设$a\in\R^{n}$,$b\in\R^{n}\setminus\left\{0\right\}$.
	\begin{enumerate}
		\item $a+\R_{\geqslant 0}b$是$\R^{n}$的闭集,并且和$\left[0,+\infty\right)$同胚.
		\item $a+\R_{>0}b$和$\left(0,+\infty\right)$同胚,是$a+\R b$的开集,但$n\geqslant 2$时$a+\R_{>0}b$不是$\R^{n}$的开集.
	\end{enumerate}
\end{proposition}

\begin{definition}
	设$x,y\in\R^{n}$.
	\begin{itemize}
		\item 记$\left[x,y\right]\coloneq\left\{\alpha x+\beta y\in\R^{n}\middle|\alpha,\beta\geqslant 0,\alpha+\beta=1\right\}$.我们称$\left[x,y\right]$是以$x$和$y$为端点的闭线段.
		\item 设$x\neq y$.记$\left(x,y\right)\coloneq\left\{\alpha x+\beta y\in\R^{n}\middle|\alpha,\beta>0,\alpha+\beta=1\right\}$.我们称$\left(x,y\right)$是以$x$和$y$为端点的闭线段.
		\item 设$x\neq y$.记$\left(x,y\right]\coloneq\left(x,y\right)\sqcup\left\{y\right\}$.我们称$\left(x,y\right]$是在$x$处开,在$y$处闭的线段.
		\item 设$x\neq y$.记$\left[x,y\right)\coloneq\left(x,y\right)\sqcup\left\{y\right\}$.我们称$\left[x,y\right)$是在$x$处闭,在$y$处开的线段.
	\end{itemize}
\end{definition}

\begin{proposition}
	设$x,y\in\R^{n}$.
	\begin{enumerate}
		\item 当$x=y$时,$\left[x,y\right]=\left\{x\right\}$;当$x\neq y$时,$\left[x,y\right]$和$\R$中的区间$\left[0,1\right]$同胚.
		\item 设$x\neq y$,则$\left(x,y\right)$和$\R$中的区间$\left(0,1\right)$同胚.
		\item 设$x\neq y$,则$\left(x,y\right]$和$\R$中的区间$\left(0,1\right]$同胚.
		\item 设$x\neq y$,则$\left[x,y\right)$和$\R$中的区间$\left[0,1\right)$同胚.
	\end{enumerate}
\end{proposition}

\begin{proposition}
	设$f:\R^{n}\to\R$是非零多项式函数,则$\R^{n}\setminus f^{-1}\left(\left\{0\right\}\right)$在$\R^{n}$中稠密.
\end{proposition}

\begin{corollary}
	设$p\leqslant n-1$,$V$是$\R^{n}$中的$p$维仿射线性簇,则$\R^{n}\setminus V$在$\R^{n}$中稠密.
\end{corollary}

\begin{proposition}
	设$H\subseteq\R^{n}$是超平面,则存在非零多项式函数$L:\R^{n}\to\R$使得$H=L^{-1}\left(\left\{0\right\}\right)$.
\end{proposition}

\begin{definition}
	设$H\subseteq\R^{n}$是超平面,非零多项式函数$L:\R^{n}\to\R$满足$H=L^{-1}\left(\left\{0\right\}\right)$.
	\begin{itemize}
		\item $L^{-1}\left(\left(0,+\infty\right)\right)$和$L^{-1}\left(\left(-\infty,0\right)\right)$称为由$H$确定的开半空间.
		\item $L^{-1}\left(\left[0,+\infty\right)\right)$和$L^{-1}\left(\left(-\infty,0\right]\right)$称为由$H$确定的闭半空间.
	\end{itemize}
\end{definition}

\begin{proposition}
	设$H\subseteq\R^{n}$是超平面,非零多项式函数$L:\R^{n}\to\R$满足$H=L^{-1}\left(\left\{0\right\}\right)$.
	\begin{enumerate}
		\item $\R^{n}\setminus H$的连通分支有且仅有$L^{-1}\left(\left(0,+\infty\right)\right)$和$L^{-1}\left(\left(-\infty,0\right)\right)$.
		\item $L^{-1}\left(\left(0,+\infty\right)\right)$和$L^{-1}\left(\left(-\infty,0\right)\right)$在$\R^{n}$中的闭包分别是$L^{-1}\left(\left(0,+\infty\right)\right)\cup H$和$L^{-1}\left(\left(-\infty,0\right)\right)\cup H$.
		\item $L^{-1}\left(\left(0,+\infty\right)\right)$和$L^{-1}\left(\left(-\infty,0\right)\right)$是$\R^{n}$的开集,二者都与$\R_{>0}$同胚.
		\item $L^{-1}\left(\left[0,+\infty\right)\right)$和$L^{-1}\left(\left(-\infty,0\right]\right)$是$\R^{n}$的闭集.
	\end{enumerate}
\end{proposition}
