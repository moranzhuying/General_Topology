\section{Grassmann流形}

\begin{definition}{Grassmann流形}{}
	给$\mathbf{G}_{n+1,p+1}\left(\R\right)$装备$\mathbb{P}\left(\bigwedge^{p+1}\left(\R^{n+1}\right)\right)$诱导的子空间拓扑,得到的拓扑空间称为Grassmann流形.
\end{definition}

\begin{proposition}{}{}
	$\mathbf{G}_{n+1,p+1}\left(\R\right)$同胚于$\mathbb{P}_{n,p}\left(\R\right)$.
\end{proposition}

\begin{corollary}{}{}
	$\mathbf{G}_{n+1,p+1}\left(\R\right)$和$\mathbf{G}_{n+1,n-p}\left(\R\right)$同胚.
\end{corollary}

\begin{corollary}{}{}
	$\mathbb{P}_{n,p}\left(\R\right)$和$\mathbb{P}_{n,n-p-1}\left(\R\right)$同胚.
\end{corollary}
