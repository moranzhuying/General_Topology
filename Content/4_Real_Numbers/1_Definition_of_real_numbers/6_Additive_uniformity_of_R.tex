\section{\texorpdfstring{$\R$}{R}的加法一致结构}

\begin{proposition}{实直线加法群的邻里基的构造}{construction-of-entourage-basis-of-real-line-addition-group}
	对各$a\in\R_{>0}$,记$U_{a}=\left\{\left(x,y\right)\in\R\times\R\middle|\left|x-y\right|<a\right\}$,那么$\left\{U_{a}\middle|a\in\R_{>0}\right\},\left\{
	U_{\frac{1}{n}}\middle|n\geq 1\right\}$都是实直线加法群的邻里基.
\end{proposition}

\begin{definition}{$\R$的加法一致结构}{}
	沿用\cref{prop:construction-of-entourage-basis-of-real-line-addition-group}的设定.实直线加法群上以$\left\{U_{a}\middle|a\in\R_{>0}\right\}$为邻里基的一致结构称为$\R$的(标准)加法一致结构.
\end{definition}

\begin{proposition}{}{}
	\begin{enumerate}
		\item $\R\to\R,x\mapsto\left|x\right|,\R\to\R,x\mapsto x^{+},\R\to\R,x\mapsto x^{-}$都一致连续.
		\item $\R\times\R\to\R,\left(x,y\right)\mapsto\sup\left(x,y\right)$和$\R\times\R\to\R,\left(x,y\right)\mapsto\inf\left(x,y\right)$都一致连续.
	\end{enumerate}
\end{proposition}

\begin{remark}{}{}
	这里提到的五个函数都可以从$\Q$到$\Q$(或$\Q\times\Q$到$\Q$)的映射连续延拓而来.
\end{remark}
