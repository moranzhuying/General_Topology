\section{实数的\texorpdfstring{$n$}{n}次方根}

本小节中,对任一$n\in\Z_{\geqslant 1}$,记$f_{n}:\R_{>0}\to\R_{>0},x\mapsto x^{n}$.

\begin{proposition}{}{}
	对任一$n\in\Z_{\geqslant 1}$,$f_{n}$是严格单调递增的连续双射,从而是同胚.
\end{proposition}

\begin{definition}{}{}
	对任一$x\in\R_{>0}$和$n\in\Z_{\geqslant 1}$,定义$x^{\frac{1}{n}}=\sqrt[n]{x}=f_{n}^{-1}\left(x\right)$称为$x$的$n$次方根.
\end{definition}

\begin{proposition}{}{}
	对任一$n\in\Z_{\geqslant 1}$,$f_{n}^{-1}$是$\R_{>0}$的拓扑群自同构.
\end{proposition}
