\chapter{实数域}

\section{\texorpdfstring{$\R$}{R}上的乘法}

\begin{lemma}{}{}
	对$0$在$\Q$中的每个邻域$V$,映射$\Q\setminus V\to\Q,x\mapsto x^{-1}$一致连续..
\end{lemma}

\begin{lemma}{}{}
	设$\mathfrak{F}$是$\Q^{\times}$的加法Cauchy滤子.若$0$不是$\mathfrak{F}$的聚点,则$\mathfrak{F}$在$\Q^{\times}\to\Q,x\mapsto x^{-1}$下的前推是加法Cauchy滤子.
\end{lemma}

\begin{proposition}{}{}
	$\Q$是拓扑域.
\end{proposition}

\begin{proposition}{}{}
	\begin{enumerate}
		\item $\Q\times\Q\to\Q,\left(x,y\right)\mapsto xy$能连续延拓为$\R\times\R\to\R,\left(x,y\right)\mapsto xy$.
		\item $\Q^{\times}\to\Q,x\mapsto x^{-1}$能连续延拓为$\R^{\times}\to\R,x\mapsto x^{-1}$.
		\item 拓扑群$\R$在连续延拓下的乘法和取逆映射下构成拓扑域.
	\end{enumerate}
\end{proposition}

\begin{definition}{实数域}{}
	我们称拓扑域$\R$为实数域.
\end{definition}
