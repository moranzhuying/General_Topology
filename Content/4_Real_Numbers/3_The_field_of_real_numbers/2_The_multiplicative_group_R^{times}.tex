\section{乘法群\texorpdfstring{$\R^{\times}$}{R*}}

\begin{proposition}{$\R^{\times}$是完备的局部紧拓扑群}{}
	$\R^{\times}$是局部紧拓扑群,从而是完备拓扑群.
\end{proposition}

\begin{proposition}{}{}
	$\R$是有序域,$\R^{\times}$是全序群.
\end{proposition}

\begin{proposition}{}{}
	$\R^{\times}$作为拓扑群同构于$\R_{>0}\times\left\{-1,1\right\}$.
\end{proposition}

\begin{definition}{}{}
	我们称$\sgn:\R^{\times}\to\left\{-1,1\right\},x\mapsto x\left|x\right|^{-1}$是符号函数.
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item $\sgn$既是满射又是拓扑群同态,每个$x\in\R^{\times}$都能唯一地表示为$\left|x\right|\sgn\left(x\right)$.
		\item 我们一般补充定义$\sgn\left(0\right)=0$,把$\sgn$延拓为$\R\to\left\{-1,0,1\right\},x\mapsto
		\begin{cases}
			1,&x>0\\
			0,&x=0\\
			-1,&x<0
		\end{cases}$.
	\end{enumerate}
\end{remark}
