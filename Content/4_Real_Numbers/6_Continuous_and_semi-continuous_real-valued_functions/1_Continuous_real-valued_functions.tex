\chapter{连续实值函数,半连续实值函数}

未加声明时,默认$X$是拓扑空间,$f,g\in\overline{\R}^{X}$.

\section{连续实值函数}

\begin{theorem}{}{}
	设$X$是非空紧空间,$f$连续,则存在$a,b\in X$使得$f\left(a\right)=\sup\limits_{x\in X}f\left(x\right)$且$b=\inf\limits_{x\in X}f\left(x\right)$.
\end{theorem}

\begin{corollary}{}{}
	设$X$是非空紧空间,$f$是连续的有限实值函数,则$f$在$X$上有界.
\end{corollary}

\begin{theorem}{}{}
	设$X$是连通空间,$f$连续,则对任一$a,b\in X$,$\alpha\in\left[\min\left\{a,b\right\},\max\left\{a,b\right\}\right]$,存在$c\in X$使得$f\left(c\right)=X$.
\end{theorem}
