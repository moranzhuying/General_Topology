\section{半连续函数}

\begin{definition}{}{}
	设$a\in X$.
	\begin{itemize}
		\item 若$\forall h\in\left(-\infty,f\left(a\right)\right)\exists V\in\mathcal{N}\left(a\right)\left(f\left(V\right)\subseteq\left(h,+\infty\right]\right)$,则称$f$在$a$处上半连续.
		\item 若$\forall h\in\left(f\left(a\right),+\infty\right)\exists V\in\mathcal{N}\left(a\right)\left(f\left(V\right)\subseteq\left[-\infty,h\right)\right)$,则称$f$在$a$处下半连续.
	\end{itemize}
\end{definition}

\begin{definition}{}{}
	若对任一$x\in X$,$f$在$x$处上半连续(下半连续),则称$f$在$X$上是上半连续(下半连续)的.
\end{definition}

\begin{proposition}{}{}
	设$a\in X$.
	\begin{enumerate}
		\item $f$在$a$处连续$\iff$ $f$在$a$处既上半连续又下半连续.
		\item $f$在$a$处下半连续$\iff$ $-f$在$a$处上半连续.
		\item 若$\exists V\in\mathcal{N}\left(a\right)\left(f\left(V\right)\subseteq\left[-\infty,f\left(a\right)\right]\right)$,则$f$在$a$处上半连续.特别地,若$f\left(a\right)=+\infty$,则$f$在$a$处上半连续.
		\item 若$\exists V\in\mathcal{N}\left(a\right)\left(f\left(V\right)\subseteq\left[f\left(a\right),+\infty\right]\right)$,则$f$在$a$处下半连续.特别地,若$f\left(a\right)=-\infty$,则$f$在$a$处下半连续.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	$f$下半连续$\iff$对任一$k\in\R$,$f^{-1}\left(\left(k,+\infty\right)\right)$是$X$的开集$\iff$对任一$k\in\R$,$f^{-1}\left(\left(-\infty,k\right]\right)$是$X$的闭集.
\end{proposition}

\begin{corollary}{}{}
	若存在$\R$的稠密子集$D$,对任一$k\in D$,$f^{-1}\left(\left(k,+\infty\right)\right)$是开集,则$f$下半连续.
\end{corollary}

\begin{corollary}{}{}
	是$A\subseteq X$,$\chi_{A}$是$A$的特征函数,则$A$是$X$的开集(闭集)$\iff$ $\chi_{A}$上半连续(下半连续).
\end{corollary}

\begin{theorem}{}{}
	设$X$是非空紧空间,$f$下半连续,则存在$a\in X$使得$f\left(a\right)=\inf\limits_{x\in X}f\left(x\right)$.
\end{theorem}

\begin{corollary}{}{}
	设$X$是非空紧空间,$f$下半连续且$f\left(X\right)\subseteq\left(-\infty,+\infty\right]$,则$f$在$X$上有下界.
\end{corollary}

\begin{proposition}{}{}
	设$f$,$g$在$a$处下半连续.
	\begin{enumerate}
		\item $\sup\left(f,g\right)$和$\inf\left(f,g\right)$在$a$处下半连续.
		\item 若$f+g$有定义,则$f+g$在$a$处下半连续.
		\item 若$\forall x\in X\left(f\left(x\right),g\left(x\right)\geqslant 0\right)$且$fg$有定义,则$fg$在$a$处下半连续.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	设$f$在$a$处下半连续且$\forall x\in X\left(f\left(x\right)\geqslant 0\right)$,则$\frac{1}{f}$在$a$处上半连续.
\end{proposition}

\begin{theorem}{}{}
	设$\left(f_{i}\right)_{i\in I}$是$\overline{\R}^{X}$的元素族,并且对任一$i\in I$,$f_{i}$在$a$处下半连续,则$\left(f_{i}\right)_{i\in I}$的上包络在$a$处下半连续.
\end{theorem}

\begin{corollary}{}{}
	设$\left(f_{i}\right)_{i\in I}$是$\overline{\R}^{X}$的元素族,并且对任一$i\in I$,$f_{i}$连续,则$\left(f_{i}\right)_{i\in I}$的上包络连续.
	
	进一步,若$I$是有限集,则$\left(f_{i}\right)_{i\in I}$的下包络下半连续.
\end{corollary}

\begin{proposition}{}{}
	$f$在$a$处下半连续$\iff$ $\varliminf_{x\to a}f\left(x\right)=f\left(a\right)$ $\iff$ $\varliminf_{x\to a}f\left(x\right)\geqslant f\left(a\right)$.
\end{proposition}

\begin{definition}{}{}
	设$A$是$X$的稠密子集,$f\in\overline{\R}^{A}$.定义
	\begin{align*}
		&g_{1}:X\to\overline{\R},x\mapsto\limit_{y\to x}\inf\limits_{y\in A}f\left(y\right),\\
		&g_{2}:X\to\overline{\R},x\mapsto\limit_{y\to x}\sup\limits_{y\in A}f\left(y\right).
	\end{align*}
	容易验证$g_{1}$下半连续,$g_{2}$上半连续.我们分别称$g_{1}$和$g_{2}$是$f$的下半连续正则化与上半连续正则化.
\end{definition}

\begin{proposition}{}{}
	设$A$是$X$的稠密子集,$f\in\overline{\R}^{A}$.记
	\begin{align*}
		W_{1}=\left\{\varphi\in\overline{\R}^{X}\middle|\varphi|_{A}\leqslant f\right\},W_{2}=\left\{\varphi\in\overline{\R}^{X}\middle|\varphi|_{A}\geqslant f\right\},
	\end{align*}
	则$W_{1}$有最大元,$W_{2}$有最小元,并且$\max W_{1}$是$f$的下半连续正则化,$\min W_{2}$是$f$的上半连续正则化.
\end{proposition}

\begin{proposition}{}{}
	设$A$是$X$的稠密子集,$f\in\overline{\R}^{A}$.
	\begin{enumerate}
		\item 若$f$下半连续,则$f$的下半连续正则化是$f$在$X$上的延拓.
		\item 若$f$上半连续,则$f$的上半连续正则化是$f$在$X$上的延拓.
	\end{enumerate}
\end{proposition}
