\section{实值函数族的包络}

未加声明时,$X$是非空集合,$\left(f_{i}\right)_{i\in I}$是$\overline{\R}^{X}$的元素族.

\begin{definition}{}{}
	$\sup\limits_{i\in I}f_{i}:X\to\overline{\R},x\mapsto\sup\limits_{i\in I}f_{i}\left(x\right)$和$\inf\limits_{i\in I}f_{i}:X\to\overline{\R},x\mapsto\inf\limits_{i\in I}f_{i}\left(x\right)$依次称为$\left(f_{i}\right)_{i\in I}$的上包络与下包络.
\end{definition}

\begin{proposition}{}{}
	在$\overline{\R}^{X}$中,$\sup\limits_{i\in I}f_{i}=\limit_{H\in\fin\left(X\right)}\sup\limits_{i\in H}f_{i}$,$\inf\limits_{i\in I}f_{i}=\limit_{H\in\fin\left(X\right)}\inf\limits_{i\in H}f_{i}$.
\end{proposition}

\begin{definition}{}{}
	若$\exists a\in\R\forall i\in I\forall x\in X\left(f_{i}\left(x\right)\leqslant a\right)$,则称$\left(f_{i}\right)_{i\in I}$在$X$上一致有上界.
	
	若$\exists a\in\R\forall i\in I\forall x\in X\left(f_{i}\left(x\right)\leqslant a\right)$,则称$\left(f_{i}\right)_{i\in I}$在$X$上一致有上界.
	
	若$\left(f_{i}\right)_{i\in I}$在$X$上一致有上界且一致有下界,则称$\left(f_{i}\right)_{i\in I}$在$X$上一致有界.
\end{definition}

\begin{proposition}{}{}
	$\left(f_{i}\right)_{i\in I}$在$X$上一致有界$\iff$ $\exists a\in\R_{\geqslant 0}\forall i\in I\forall x\in X\left(\left|f_{i}\left(x\right)\right|\leqslant a\right)$.
\end{proposition}
