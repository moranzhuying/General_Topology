\section{实值函数相对滤子的上下极限}

记号约定:$E$是装备了滤子$\mathfrak{F}$的集合.

\begin{definition}{函数沿滤子的上极限和下极限}{}
	设$f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$.
	\begin{enumerate}
		\item 我们称$\varlimsup\limits_{\mathfrak{F}}f\coloneq\limit_{H\in\mathfrak{F}}\sup\limits_{x\in H}f\left(x\right)$是$f$沿$\mathfrak{F}$的上极限.
		\item 我们称$\varliminf\limits_{\mathfrak{F}}f\coloneq\limit_{H\in\mathfrak{F}}\inf\limits_{x\in H}f\left(x\right)$是$f$沿$\mathfrak{F}$的上极限.
	\end{enumerate}
\end{definition}

\begin{proposition}{}{}
	设$f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$.
	\begin{enumerate}
		\item $\inf\limits_{x\in E}f\left(x\right)\leq\varliminf\limits_{\mathfrak{F}}f\leq\varlimsup\limits_{\mathfrak{F}}f\leq\sup\limits_{x\in E}f\left(x\right)$.
		\item $\varlimsup\limits_{\mathfrak{F}}f=\inf\limits_{M\in\mathfrak{F}}\sup\limits_{x\in M}f\left(x\right),\varliminf\limits_{\mathfrak{F}}f=\sup\limits_{M\in\mathfrak{F}}\inf\limits_{x\in M}f\left(x\right)$.
		\item $\varliminf\limits_{\mathfrak{F}}f=-\varlimsup\limits_{\mathfrak{F}}\left(-f\right)$.
	\end{enumerate}
\end{proposition}

\begin{theorem}{}{}
	设$f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$,那么$\varlimsup\limits_{\mathfrak{F}}f$是$f$沿$\mathfrak{F}$的最大聚点.
\end{theorem}

\begin{corollary}{}{}
	设$f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$.
	\begin{enumerate}
		\item $\varliminf\limits_{\mathfrak{F}}f=\varlimsup\limits_{\mathfrak{F}}f$ $\iff$ $f$沿$\mathfrak{F}$收敛.两条件之一成立时,$\varlimsup\limits_{\mathfrak{F}}f=\varliminf\limits_{\mathfrak{F}}f=\limit_{\mathfrak{F}}f$.
		\item 如果$\mathfrak{G}$是$E$上比$\mathfrak{F}$粗的滤子,那么$\varlimsup\limits_{\mathfrak{F}}f\leq\varlimsup\limits_{\mathfrak{G}}f\leq\varlimsup\limits_{\mathfrak{G}}f\leq\varlimsup\limits_{\mathfrak{F}}f$.进一步,如果$f$沿$\mathfrak{G}$收敛,那么
		\begin{align*}
			\varliminf_{\mathfrak{G}}f\leq\limit_{\mathfrak{G}}f\leq\varlimsup\limits_{\mathfrak{G}}f.
		\end{align*}
		\item 如果$A\in\mathfrak{F}$,$\mathfrak{F}_{A}$是$\mathfrak{F}$在$A$上诱导的滤子,$f_{A}=f|_{A}$,那么$\varlimsup\limits_{\mathfrak{F}_{A}}f_{A}=\varlimsup\limits_{\mathfrak{F}}f$.
	\end{enumerate}
\end{corollary}

\begin{proposition}{}{}
	设$f,g\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$.如果$f\leq g$,那么$\varlimsup\limits_{\mathfrak{F}}f\leq\varlimsup\limits_{\mathfrak{F}}g,\varliminf\limits_{\mathfrak{F}}f\leq\varliminf\limits_{\mathfrak{F}}g$.
\end{proposition}

本节接下来的内容中,默认$E$是拓扑空间.

\begin{definition}{拓扑空间中某一点的上下极限}{}
	设$a\in X,f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$.
	\begin{enumerate}
		\item 我们称$\varlimsup\limits_{x\to a}f\left(x\right)\coloneq\varlimsup\limits_{\mathcal{N}\left(a\right)}f$是$f$在$a$处的上极限.
		\item 我们称$\varliminf\limits_{x\to a}f\left(x\right)\coloneq\varliminf\limits_{\mathcal{N}\left(a\right)}f$是$f$在$a$处的下极限.
	\end{enumerate}
\end{definition}

\begin{remark}{}{}
	根据定义直接得到$\varliminf\limits_{x\to a}f\left(x\right)\leq f\left(a\right)\leq\varlimsup\limits_{x\to a}f\left(x\right)$.
\end{remark}

\begin{definition}{拓扑子空间中某一点的上下极限}{}
	设$A$是$E$的拓扑子空间,$a\in\closure\left(A\right),f\in\Hom_{\mathsf{Set}}\left(A,\overline{\R}\right)$.记$\mathcal{N}_{A}\left(a\right)$是$a$在$A$中的邻域滤子.
	\begin{enumerate}
		\item 我们称$\varlimsup\limits_{\substack{x\to a\\ x\in A}}f\left(x\right)\coloneq\inf\limits_{U\in\mathcal{N}_{A}\left(a\right)}\sup\limits_{x\in U\cap A}f\left(x\right)$是$f$在$A$中趋于$a$时的上极限.
		\item 我们称$\varliminf\limits_{\substack{x\to a\\ x\in A}}f\left(x\right)\coloneq\sup\limits_{U\in\mathcal{N}_{A}\left(a\right)}\inf\limits_{x\in U\cap A}f\left(x\right)$是$f$在$A$中趋于$a$时的下极限.
	\end{enumerate}
\end{definition}

\begin{definition}{拓扑子空间中某一点的去心上下极限}{}
	设$A$是$E$的拓扑子空间,$a\in\closure\left(A\setminus\left\{a\right\}\right),f\in\Hom_{\mathsf{Set}}\left(A,\overline{\R}\right)$.记$\mathcal{N}_{A}\left(a\right)$是$a$在$A$中的邻域滤子.
	\begin{enumerate}
		\item 我们称$\varlimsup\limits_{\substack{x\to a\\ x\in A\setminus\left\{a\right\}}}f\left(x\right)\coloneq\inf\limits_{U\in\mathcal{N}_{A}\left(a\right)}\sup\limits_{x\in(U\cap A)\setminus\left\{a\right\}}f\left(x\right)$是$f$在$A$中趋于$a$时的去心上极限.
		\item 我们称$\varliminf\limits_{\substack{x\to a\\ x\in A\setminus\left\{a\right\}}}f\left(x\right)\coloneq\sup\limits_{U\in\mathcal{N}_{A}\left(a\right)}\inf\limits_{x\in(U\cap A)\setminus\left\{a\right\}}f\left(x\right)$是$f$在$A$中趋于$a$时的去心下极限.
	\end{enumerate}
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 如果$A$是$E$的子空间,$a\in\closure\left(A\right),f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$,那么
		\begin{align*}
			\limit_{x\to a}\inf\limits_{x\in E}f\left(x\right)\leq\limit_{x\to a}\inf\limits_{x\in A}f\left(x\right)\leq\limit_{x\to a}\inf\limits_{x\in A}f\left(x\right)\leq\limit_{x\to a}\sup\limits_{x\in E}f\left(x\right).
		\end{align*}
		\item 如果$V\in\mathcal{N}\left(a\right),f\in\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$,那么$\limit_{x\to a}\sup\limits_{x\in V\cap E}f\left(x\right)=\limit_{x\to a}\sup\limits_{x\in E}f\left(x\right)$.
	\end{enumerate}
\end{remark}

\begin{definition}{序列的上下极限}{}
	设$\left(x_{n}\right)_{n=0}^{\infty}$是$\overline{\R}$中的序列,$\mathfrak{G}$是$\N$的\text{Fr\'{e}net}滤子.
	\begin{enumerate}
		\item 我们称$\varlimsup\limits_{n\to +\infty}x_{n}\coloneq\varlimsup\limits_{\mathfrak{G}}x_{n}$是$\left(x_{n}\right)_{n=0}^{\infty}$的上极限.
		\item 我们称$\varliminf\limits_{n\to +\infty}x_{n}\coloneq\varliminf\limits_{\mathfrak{G}}x_{n}$是$\left(x_{n}\right)_{n=0}^{\infty}$的下极限.
	\end{enumerate}
\end{definition}

\begin{proposition}{}{}
	设$\left(f_{n}\right)_{n=0}^{\infty}$是$\Hom_{\mathsf{Set}}\left(E,\overline{\R}\right)$中的序列,那么
	\begin{gather*}
		\varlimsup\limits_{n\to +\infty}f_{n}=\inf\limits_{n\geq 0}\sup\limits_{m\geq n}f_{m}=\limit_{n\to +\infty}\sup\limits_{m\geq m}f_{m},\\
		\varliminf\limits_{n\to +\infty}f_{n}=\sup\limits_{n\geq 0}\inf\limits_{m\geq n}f_{m}=\limit_{n\to +\infty}\inf\limits_{m\geq m}f_{m}.
	\end{gather*}
\end{proposition}
