\section{实值函数相对滤子的上下极限}

未加声明时,$X$是非空集合,$\mathcal{F}$是$X$上的滤子,$f,g\in\overline{\R}^{X}$,$n\in\Z_{\geqslant 1}$,$\left(f_{i}\right)_{i=1}^{n}$是$\overline{\R}^{X}$的元素族.

\begin{definition}{}{}
	设$f\in\overline{\R}^{X}$.
	\begin{itemize}
		\item $\varlimsup_{\mathcal{F}}f\coloneq\limit_{M\in\fin\left(X\right)}\sup\limits_{x\in M}f\left(x\right)$称为$f$关于$\mathcal{F}$的上极限.
		\item $\varliminf_{\mathcal{F}}f\coloneq\limit_{M\in\fin\left(X\right)}\inf\limits_{x\in M}f\left(x\right)$称为$f$关于$\mathcal{F}$的下极限.
	\end{itemize}
\end{definition}

\begin{proposition}{}{}
	设$f\in\overline{\R}^{X}$,则$\varlimsup_{\mathcal{F}}f=\inf\limits_{M\in\mathcal{F}}\sup\limits_{x\in M}f\left(x\right)$,$\varliminf_{\mathcal{F}}f=\sup\limits_{M\in\mathcal{F}}\inf\limits_{x\in M}f\left(x\right)$.
\end{proposition}

\begin{proposition}{}{}
	$\inf\limits_{x\in X}f\left(x\right)\leqslant\varliminf_{\mathcal{F}}f\leqslant\varlimsup_{\mathcal{F}}f\leqslant\sup\limits_{x\in X}f\left(x\right)$,$\varliminf_{\mathcal{F}}f=-\varlimsup_{\mathcal{F}}\left(-f\right)$.
\end{proposition}

\begin{theorem}{}{}
	$\varlimsup_{\mathcal{F}}f$等于$f$关于$\mathcal{F}$在$\overline{\R}$中的最大聚点;$\varliminf_{\mathcal{F}}f$等于$f$关于$\mathcal{F}$在$\overline{\R}$中的最小聚点.
\end{theorem}

\begin{corollary}{}{}
	$f$关于$\mathcal{F}$在$\overline{\R}$中有极限点$\iff$ $\varlimsup_{\mathcal{F}}f=\varliminf_{\mathcal{F}}f$.两个等价条件之一成立时,$\varlimsup_{\mathcal{F}}f=\varliminf_{\mathcal{F}}f=\limit_{\mathcal{F}}f$.
\end{corollary}

\begin{corollary}{}{}
	设$\mathcal{G}$是$X$上比$\mathcal{F}$粗的滤子,则$\varliminf_{\mathcal{F}}f\leqslant\varliminf_{\mathcal{G}}f\leqslant\varlimsup_{\mathcal{G}}f\leqslant\varlimsup_{\mathcal{F}}f$.
\end{corollary}

\begin{corollary}{}{}
	设$A\in\mathcal{F}$,$\mathcal{F}_{A}$是$\mathcal{F}$在$A$上诱导的滤子,则$\varlimsup_{\mathcal{F}_{A}}f|_{A}=\varlimsup_{\mathcal{F}}f$,$\varliminf_{\mathcal{F}_{A}}f|_{A}=\varliminf_{\mathcal{F}}f$
\end{corollary}

\begin{proposition}{}{}
	$\exists M\in\mathcal{F}\forall x\in M\left(f\left(x\right)\leqslant g\left(x\right)\right)$,则$\varlimsup_{\mathcal{F}}f\leqslant\varlimsup_{\mathcal{F}}g$,$\varliminf_{\mathcal{F}}f\leqslant\varliminf_{\mathcal{F}}g$.
\end{proposition}

\begin{definition}{}{}
	设$X$是拓扑空间,$a\in X$.定义$\varlimsup_{x\to a}f\left(x\right)=\varlimsup_{\mathcal{V}\left(a\right)}f$,$\varliminf_{x\to a}f\left(x\right)=\varliminf_{\mathcal{V}\left(a\right)}f.$
\end{definition}

\begin{definition}{}{}
	设$X$是拓扑空间,$\emptyset\neq A\subseteq X$,$a\in\closure\left(A\right)$.定义
	\begin{align*}
		\varlimsup_{\substack{x\to a\\x\in A}}f\left(x\right)\coloneq\varlimsup_{\mathcal{F}}f,\varliminf_{\substack{x\to a\\x\in A}}f\left(x\right)=\varliminf_{\mathcal{V}\left(a\right)}f.
	\end{align*}
	分别称$\varlimsup_{\substack{x\to a\\x\in A}}f\left(x\right)$和$\varliminf_{\substack{x\to a\\x\in A}}f\left(x\right)$是$f$在$a$处的上极限和下极限.
\end{definition}

\begin{proposition}{}{}
	设$X$是拓扑空间,$a\in X$.对任一$V\in\mathcal{N}\left(a\right)$,$\varlimsup_{\substack{x\to a\\x\in V}}f\left(x\right)=\sup\limits_{x\in V}f\left(x\right)$,$\varliminf_{\substack{x\to a\\x\in V}}f\left(x\right)=\inf\limits_{x\in V}f\left(x\right)$.
\end{proposition}

\begin{definition}{}{}
	设$X$是拓扑空间,$\emptyset\neq A\subseteq X$,$a\in\closure\left(A\setminus\left\{a\right\}\right)$,$f\in\overline{\R}^{A}$.定义
	\begin{align*}
		\varlimsup_{\substack{x\to a\\x\in A}}f\left(x\right)\coloneq\inf\limits_{V\in\mathcal{N}\left(a\right)}\sup\limits_{x\in\left(A\cap V\right)\setminus\left\{a\right\}}f\left(x\right),
		\varliminf_{\substack{x\to a\\x\in A}}f\left(x\right)\coloneq\sup\limits_{V\in\mathcal{N}\left(a\right)}\inf\limits_{x\in\left(A\cap V\right)\setminus\left\{a\right\}}f\left(x\right).
	\end{align*}
	分别称$\varlimsup_{\substack{x\to a\\x\in A}}f\left(x\right)$和$\varliminf_{\substack{x\to a\\x\in A}}f\left(x\right)$是$f$沿$A$在$a$处的上极限和下极限.
\end{definition}

\begin{definition}{}{}
	设$\left(x_{n}\right)_{n=0}^{\infty}$是$\overline{\R}$中的序列,$\mathcal{H}$是$\Z_{\geqslant 0}$中的\text{Fr\'{e}net}滤子.定义
	\begin{align*}
		\varlimsup_{n\to+\infty}x_{n}=\varlimsup_{\mathcal{H}}x_{n},\varliminf_{n\to+\infty}x_{n}=\varliminf_{\mathcal{H}}x_{n}.
	\end{align*}
	分别称$\varlimsup_{n\to+\infty}x_{n}$和$\varliminf_{n\to+\infty}x_{n}$是$\left(x_{n}\right)_{n=0}^{\infty}$的上确界和下确界.
\end{definition}

\begin{definition}{}{}
	设$\left(g_{n}\right)_{n=0}^{\infty}$是$\overline{\R}^{X}$的元素族.定义
	\begin{align*}
		&\varlimsup_{n\to+\infty}g_{n}:X\to\overline{\R},x\mapsto\varlimsup_{n\to+\infty}g_{n}\left(x\right),\\
		&\varliminf_{n\to+\infty}g_{n}:X\to\overline{\R},x\mapsto\varliminf_{n\to+\infty}g_{n}\left(x\right).
	\end{align*}
	分别称$\varlimsup_{n\to+\infty}f_{n}$和$\varliminf_{n\to+\infty}f_{n}$是$\left(g_{n}\right)_{n=0}^{\infty}$的上确界和下确界.
\end{definition}

\begin{proposition}{}{}
	设$\left(g_{n}\right)_{n=0}^{\infty}$是$\overline{\R}^{X}$的元素族,则
	\begin{align*}
		\varlimsup_{n\to+\infty}g_{n}=\inf_{n\geqslant 0}\sup_{m\geqslant n}g_{m}=\limit_{n\to+\infty}\sup\limits_{m\geqslant n}g_{m},\varliminf_{n\to+\infty}g_{n}=\sup_{n\geqslant 0}\inf_{m\geqslant n}g_{m}=\limit_{n\to+\infty}\inf\limits_{m\geqslant n}g_{n}.
	\end{align*}
\end{proposition}
