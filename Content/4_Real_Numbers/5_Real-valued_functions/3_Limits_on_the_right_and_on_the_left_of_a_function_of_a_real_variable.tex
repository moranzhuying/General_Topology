\section{单变量实值函数在某点的左右极限}

记号约定:$A$是$\overline{\R}$的非空子集,$X$是拓扑空间,$a\in\overline{\R}$.

\begin{definition}{实值函数的左右极限}{}
	设$f\in\Hom_{\mathsf{Set}}\left(A,\overline{\R}\right)$.
	\begin{enumerate}
		\item 设$\mathfrak{F}$是$A\cap\left[-\infty,a\right)$的截面滤子,$a\neq -\infty$且$a\in\closure\left(A\cap\left[-\infty,a\right)\right)$.如果$f$沿$\mathfrak{F}$收敛,则称$f$的极限为$f$在$a$处关于$A$的左极限,当$X$是Hausdorff空间时将该极限记作$\limit_{\substack{x\to a\\x\in X\cap\left(-\infty,a\right)}}f\left(x\right)$或$f\left(a^{-}\right)$.
		\item 设$\mathfrak{F}$是$X\cap\left(a,+\infty\right]$的截面滤子,$a\neq -\infty$且$a\in\closure\left(X\cap\left(a,+\infty\right]\right)$.如果$f$沿$\mathfrak{F}$收敛,则称$f$的极限为$f$在$a$处关于$A$的右极限,当$X$是Hausdorff空间时将该极限记作$=\limit_{\substack{x\to a\\x\in X\cap\left(a,+\infty\right)}}f\left(x\right)$或$f\left(a^{+}\right)$.
	\end{enumerate}
\end{definition}

\begin{proposition}{}{}
	设$A\subseteq\overline{\R},f\in\Hom_{\mathsf{Set}}\left(A,\overline{\R}\right)$.
	\begin{enumerate}
		\item 设$a\neq -\infty,a\in A\cap\left[-\infty,a\right)$.如果$f$单调递增,那么$f$在$a$处相对$A$有左极限.
		\item 设$a\neq +\infty,a\in A\cap\left(a,+\infty\right]$.如果$f$单调减,那么$f$在$a$处相对$A$有右极限.
	\end{enumerate}
\end{proposition}
