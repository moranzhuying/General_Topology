\chapter{实值函数}

\section{实值函数}

记号约定:$X$是集合.

\begin{definition}{实值函数,有限实值函数}{}
	我们称从$X$到$\R$的映射是$X$上的有限实值函数,称从$X$到$\overline{\R}$的函数是$X$上的实值函数.
\end{definition}

\begin{proposition}{}{}
	给$\Hom_{\mathsf{Set}}\left(X,\R\right)$装备逐点序.
	\begin{enumerate}
		\item $\Hom_{\mathsf{Set}}\left(X,\R\right)$是格.
		\item 对每个$f,g\in\Hom_{\mathsf{Set}}\left(X,\R\right)$,二者在$\Hom_{\mathsf{Set}}\left(X,\R\right)$中的上确界和下确界分别是
		\begin{gather*}
			\sup\left(f,g\right):X\to\overline{\R},x\mapsto\sup\left(f\left(x\right),g\left(x\right)\right),\\
			\inf\left(f,g\right):X\to\overline{\R},x\mapsto\inf\left(f\left(x\right),g\left(x\right)\right).
		\end{gather*}
	\end{enumerate}
\end{proposition}

\begin{definition}{有(上/下)界的实值函数}{}
	设$f\in\Hom_{\mathsf{Set}}\left(X,\R\right)$.
	\begin{enumerate}
		\item 如果$\im\left(f\right)\subseteq\left[-\infty,+\infty\right]$且有上界,则称$f$在$X$上有上界.
		\item 如果$\im\left(f\right)\subseteq\left(-\infty,+\infty\right]$且有下界,则称$f$在$X$上有下界.
		\item 如果$\im\left(f\right)$是$\left(-\infty,+\infty\right)$的有界子集,则称$f$在$X$上有界.
	\end{enumerate}
\end{definition}

\begin{remark}{}{}
	在$E$上有界的实值函数一定是有限实值函数,反之不成立.
\end{remark}
