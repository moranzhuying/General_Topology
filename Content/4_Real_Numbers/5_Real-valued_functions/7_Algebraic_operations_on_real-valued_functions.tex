\section{实值函数的代数运算}

记号约定:$X,X_{1},X_{2}$是集合.

\begin{definition}{}{}
	\begin{enumerate}
		\item 设$f,g\in\Hom_{\mathsf{Set}}\left(X,\overline{\R}\right)$.
		\begin{itemize}
			\item 当对每个$x\in X$,$f\left(x\right)+g\left(x\right)$有定义时,令$f+g:X\to\overline{\R},x\mapsto f\left(x\right)+g\left(x\right)$.
			\item 当对每个$x\in X$,$f\left(x\right)g\left(x\right)$有定义时,令$fg:X\to\overline{\R},x\mapsto f\left(x\right)g\left(x\right)$.
		\end{itemize}
		\item 设$f\in\Hom_{\mathsf{Set}}\left(X,\overline{\R}\right)$.当对每个$x\in X$,$\left(f\left(x\right)\right)^{-1}$有定义时,令$\frac{1}{f}:X\to\overline{\R},x\mapsto\left(f\left(x\right)\right)^{-1}$.特别地,
		\begin{itemize}
			\item 当$X=\left[0,+\infty\right]$时,补充定义$\frac{1}{f}\left(0\right)\coloneq +\infty$;
			\item 当$X=\left[-\infty,0\right]$时,补充定义$\frac{1}{f}\left(0\right)\coloneq -\infty$.
		\end{itemize}
	\end{enumerate}
\end{definition}

\begin{proposition}{}{}
	设$\mathfrak{F}$是$X$上的滤子,$A$是$X$的非空子集,$f,g\in\Hom_{\mathsf{Set}}\left(X,\overline{\R}\right)$.
	\begin{enumerate}
		\item 如果$\sup\limits_{x\in A}f\left(x\right)+\inf\limits_{x\in A}g\left(x\right),\sup\limits_{x\in A}f\left(x\right)+\sup\limits_{x\in A}g\left(x\right)$有定义,对每个$x\in A$,$f\left(x\right)+g\left(x\right)$有定义,那么
		\begin{align*}
			\sup\limits_{x\in A}f\left(x\right)+\inf\limits_{x\in A}g\left(x\right)\leq\sup\limits_{x\in A}\left(f\left(x\right)+g\left(x\right)\right)\leq\sup\limits_{x\in A}f\left(x\right)+\sup\limits_{x\in A}g\left(x\right).
		\end{align*}
		\item 如果$f|_{A},g|_{A}\geq 0$,$\sup\limits_{x\in A}f\left(x\right)\sup\limits_{x\in A}g\left(x\right),\sup\limits_{x\in A}f\left(x\right)\inf\limits_{x\in A}g\left(x\right)$有定义,对每个$x\in A$,$f\left(x\right)g\left(x\right)$有定义,那么
		\begin{align*}
			\sup\limits_{x\in A}f\left(x\right)\inf\limits_{x\in A}g\left(x\right)\leq\sup\limits_{x\in A}\left(f\left(x\right)g\left(x\right)\right)\leq\sup\limits_{x\in A}f\left(x\right)\sup\limits_{x\in A}g\left(x\right).
		\end{align*}
		\item 如果$f|_{A}\geq 0$,那么$\sup\limits_{x\in A}\left(f\left(x\right)\right)^{-1}=\left(\inf\limits_{x\in A}f\left(x\right)\right)^{-1}$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	设$f\in\Hom_{\mathsf{Set}}\left(X,\overline{\R}\right),k\in\R$,$A$是$X$的非空子集
	\begin{enumerate}
		\item 如果$k+\sup\limits_{x\in A}f\left(x\right)$有定义,对每个$x\in A$,$k+f\left(x\right)$有定义,那么$\sup\limits_{x\in A}\left(k+f\left(x\right)\right)=k+\sup\limits_{x\in A}f\left(x\right)$.
		\item 如果$k\geq 0$,$k\sup\limits_{x\in A}f\left(x\right)$有定义,对每个$x\in A$,$kf\left(x\right)$有定义,那么$\sup\limits_{x\in A}kf\left(x\right)=k\sup\limits_{x\in A}f\left(x\right)$.
	\end{enumerate}
\end{corollary}

\begin{corollary}{}{}
	设$f_{1}\in\Hom_{\mathsf{Set}}\left(X_{1},\overline{\R}\right),f_{2}\in\Hom_{\mathsf{Set}}\left(X_{2},\overline{\R}\right)$,$A_{1},A_{2}$分别是$X_{1},X_{2}$的非空子集.
	\begin{enumerate}
		\item 如果$\sup\limits_{x_{1}\in A_{1}}f_{1}\left(x_{1}\right)+\sup\limits_{x_{2}\in A_{2}}f\left(x_{2}\right)$有定义,对每个$\left(x_{1},x_{2}\right)\in A_{1}\times A_{2}$,$f_{1}\left(x_{1}\right)+f_{2}\left(x_{2}\right)$有定义,那么
		\begin{align*}
			\sup\limits_{\left(x_{1},x_{2}\right)\in A_{1}\times A_{2}}\left(f_{1}\left(x_{1}\right)+f_{2}\left(x_{2}\right)\right)=\sup\limits_{x_{1}\in A_{1}}f_{1}\left(x_{1}\right)+\sup\limits_{x_{2}\in A_{2}}f_{2}\left(x_{2}\right).
		\end{align*}
		\item 如果$f_{1}|_{A_{1}},f_{2}|_{A_{2}}\geqslant 0$,$\sup\limits_{x\in A_{1}}f_{1}\left(x\right)\sup\limits_{x_{2}\in A_{2}}f\left(x_{2}\right)$有定义,对每个$\left(x_{1},x_{2}\right)\in A_{1}\times A_{2}$,$f_{1}\left(x_{1}\right)f_{2}\left(x_{2}\right)$有定义,那么
		\begin{align*}
			\sup\limits_{\left(x_{1},x_{2}\right)\in A_{1}\times A_{2}}\left(f_{1}\left(x_{1}\right)f_{2}\left(x_{2}\right)\right)=\sup\limits_{x_{1}\in A_{1}}f_{1}\left(x_{1}\right)\sup\limits_{x_{2}\in A_{2}}f_{2}\left(x_{2}\right).
		\end{align*}
	\end{enumerate}
\end{corollary}

\begin{corollary}{}{}
	\begin{enumerate}
		\item 如果$A,B$是$\overline{\R}$的子集,$\sup\left(A\right)+\sup\left(B\right)$有定义,对每个$\left(x,y\right)\in A\times B$,$x+y$有定义,那么
		\begin{align*}
			\sup\left(A+B\right)=\sup\left(A\right)+\sup\left(B\right).
		\end{align*}
		\item 如果$A,B$是$\left[0,+\infty\right]$的子集,$\sup\left(A\right)\sup\left(B\right)$有定义,对每个$\left(x,y\right)\in A\times B$,$xy$有定义,那么
		\begin{align*}
			\sup\left(AB\right)=\sup\left(A\right)\sup\left(B\right).
		\end{align*}
	\end{enumerate}
\end{corollary}

\begin{proposition}{}{}
	设$X$是装备了滤子$\mathfrak{F}$的上定向集,$f,g\in\Hom_{\mathsf{Set}}\left(X,\overline{\R}\right)$.
	\begin{enumerate}
		\item 如果$\varlimsup\limits_{\mathfrak{F}}f+\varliminf\limits_{\mathfrak{F}}g,\varlimsup\limits_{\mathfrak{F}}f+\varlimsup\limits_{\mathfrak{F}}g,f+g$有定义,那么
		\begin{align*}
			\varlimsup\limits_{\mathfrak{F}}f+\varliminf\limits_{\mathfrak{F}}g\leq\varlimsup\limits_{\mathfrak{F}}\left(f+g\right)\leq\varlimsup\limits_{\mathfrak{F}}f+\varlimsup\limits_{\mathfrak{F}}g.
		\end{align*}
		\item 如果$f,g\geq 0$,$\varlimsup\limits_{\mathfrak{F}}f\varlimsup\limits_{\mathfrak{F}}g,\varlimsup\limits_{\mathfrak{F}}f\varliminf\limits_{\mathfrak{F}}g,fg$有定义,那么
		\begin{align*}
			\sup\limits_{\mathfrak{F}}f\inf\limits_{\mathfrak{F}}g\leq\sup\limits_{\mathfrak{F}}\left(fg\right)\leq\sup\limits_{\mathfrak{F}}f\sup\limits_{\mathfrak{F}}g.
		\end{align*}
		\item 如果$f\geq 0$,那么$\varlimsup\limits_{\mathfrak{F}}\frac{1}{f}=\left(\varliminf\limits_{\mathfrak{F}}f\right)^{-1}$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	设$X$是装备了滤子$\mathfrak{F}$的上定向集,$f,g\in\Hom_{\mathsf{Set}}\left(X,\overline{\R}\right)$.
	\begin{enumerate}
		\item 如果$g$沿$\mathfrak{F}$收敛,$f+g$有定义,$\varlimsup\limits_{\mathfrak{F}}f+\limit_{\mathfrak{F}}g$有定义,那么$\varlimsup\limits_{\mathfrak{F}}\left(f+g\right)=\varlimsup\limits_{\mathfrak{F}}f+\limit_{\mathfrak{F}}g$.
		\item 如果$g$沿$\mathfrak{F}$收敛,$f,g\geq 0$,$fg$有定义,$\varlimsup\limits_{\mathfrak{F}}f\limit_{\mathfrak{F}}g$有定义,那么$\varlimsup\limits_{\mathfrak{F}}\left(fg\right)=\varlimsup\limits_{\mathfrak{F}}f\limit_{\mathfrak{F}}g$.
		\item 如果$\limit_{\mathfrak{F}}f=+\infty,\varliminf\limits_{\mathfrak{F}}g>-\infty$,$f+g$有定义,那么$\limit_{\mathfrak{F}}\left(f+g\right)=+\infty$.
		\item 如果$\limit_{\mathfrak{F}}f=+\infty,\varliminf\limits_{\mathfrak{F}}g>0$,$fg$有定义,那么$\limit_{\mathfrak{F}}fg=+\infty$.
	\end{enumerate}
\end{corollary}
