\section{定向集上的实值函数}

记号约定:$X$是装备了滤子$\mathfrak{F}$的集合.

\begin{theorem}{不等式的延拓定理}{}
	设$f,g\in\Hom_{\mathsf{Set}}\left(X,\R\right)$,二者沿$\mathfrak{F}$收敛.
	\begin{enumerate}
		\item 如果对每个$A\in\mathfrak{F}$有$f|_{A}\leq g|_{A}$,那么$\limit_{\mathfrak{F}}f\leq\limit_{\mathfrak{F}}g$.
		\item 如果$f\leq g$,那么$\limit_{\mathfrak{F}}f\leq\limit_{\mathfrak{F}}g$.
	\end{enumerate}
\end{theorem}

\begin{remark}{}{}
	如果$f<g$,那么$\limit_{\mathfrak{F}}\leq\limit_{\mathfrak{F}}g$,但是$\limit_{\mathfrak{F}}f<\limit_{\mathfrak{F}}g$不一定成立.换言之,取极限不能保持严格的不等式.
\end{remark}

\begin{theorem}{单调极限定理}{}
	设$X$是偏序集,$A$是$X$的上定向集.
	\begin{enumerate}
		\item 每个单调函数$f:A\to\overline{\R}$在$A$中都有极限点.
		\item 如果$f:A\to\overline{\R}$递增,那么$\limit_{A}f=\sup\limits_{x\in A}f\left(x\right)$.
		\item 如果$f:A\to\overline{\R}$递减,那么$\limit_{A}f=\inf\limits_{x\in A}f\left(x\right)$.
	\end{enumerate}
\end{theorem}

\begin{corollary}{}{}
	设$X$是偏序集,$A$是$X$的上定向集,$f\in\Hom_{\mathsf{Set}}\left(A,\overline{\R}\right)$.
	\begin{enumerate}
		\item 当$f$递增时,$f$在$A$中的极限有限 $\iff$ $f$有上界.
		\item 当$f$递减时,$f$在$A$中的极限有限 $\iff$ $f$有下界.
	\end{enumerate}
\end{corollary}

\begin{proposition}{}{}
	\begin{enumerate}
		\item $\R$中的每个单调序列在$\overline{\R}$中都收敛.
		\item 设$\left(x_{n}\right)_{n=0}^{\infty}$是$\R$中的递增序列.
		\begin{itemize}
			\item 如果$\left(x_{n}\right)_{n=0}^{\infty}$有上界,那么$\left(x_{n}\right)_{n=0}^{\infty}$的极限点在$\R$中.
			\item 如果$\left(x_{n}\right)_{n=0}^{\infty}$无上界,那么$\left(x_{n}\right)_{n=0}^{\infty}$的极限点是$+\infty$.
		\end{itemize}
	\end{enumerate}
\end{proposition}

\begin{remark}{}{}
	$\N$中每个严格递增的序列都收敛到$+\infty$.
\end{remark}
