\section{扩充实直线}

\begin{proposition}{扩充实直线的构造}{}
	给$\R$并接两个新元素$-\infty,+\infty$,所得到的集合记作$\overline{\R}$,将其视同$\R\sqcup\left\{-\infty,+\infty\right\}$,相应地把$\R$视同$\overline{\R}$的子集.
	\begin{enumerate}
		\item 定义$\overline{\R}$上的关系$\leq$如下:$-\infty<+\infty$,对每个$a\in\R$有$-\infty<a<+\infty$.那么,$\left(\overline{\R},\leq\right)$是全序集.
		\item 给$\overline{\R}$装备序拓扑,那么该拓扑在$\R$上诱导的子空间拓扑就是$\R$上的标准拓扑.
	\end{enumerate}
\end{proposition}

\begin{definition}{扩充实直线}{}
	我们称拓扑空间$\overline{\R}$为扩充实直线.
\end{definition}

\begin{proposition}{}{}
	设$a,b\in\R$且$a<b$,$f\in\Hom_{\mathsf{Top}}\left(\left(a,b\right),\R\right)$.
	\begin{enumerate}
		\item $f$能唯一地延拓为$\bar{f}\in\Hom_{\mathsf{Top}}\left(\left[a,b\right],\overline{\R}\right)$.
		\item 如果$f$保序,那么$\bar{f}$是序同构.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	$\overline{\R}$是紧空间.
\end{proposition}

\begin{remark}{}{}
	由此可见,如果$a,b\in\R$且$a<b$,那么$\overline{\R}$上满足如下条件一致结构存在且唯一:
	\begin{itemize}
		\item 该一致结构与$\overline{\R}$上的序拓扑相容;
		\item 下面两个一致空间是一致同构的:给$\overline{\R}$装备该一致结构形成的一致空间,装备了$\R$的加法一致结构诱导的子一致结构的$\left[a,b\right]$所形成的一致空间.
	\end{itemize}
	注意,$\R$上的一致结构在$\R$上诱导的子一致结构不是$\R$的标准加法一致结构.
\end{remark}

\begin{proposition}{}{}
	$\overline{\R}$是完备格.
\end{proposition}

\begin{proposition}{}{}
	设$A\subseteq\overline{\R}$,则$A$是$\overline{\R}$的连通集$\iff$ $A$是$\overline{\R}$的区间.
\end{proposition}

\begin{corollary}{}{}
	$\overline{\R}$既连通又局部连通.
\end{corollary}

\begin{proposition}{}{}
	设$I$是$\overline{\R}$的区间,$f\in\Hom_{\mathsf{Top}}\left(I,\R\right)$,那么$f$严格单调$\iff$ $f^{\circ}$是同胚.两条件之一成立时,$\im\left(f\right)$是$\overline{\R}$的区间.
\end{proposition}

\begin{proposition}{}{}
	$\overline{\R}\times\overline{\R}\to\overline{\R},\left(x,y\right)\mapsto\sup\left(x,y\right)$和$\overline{\R}\times\overline{\R}\to\overline{\R},\left(x,y\right)\mapsto\inf\left(x,y\right)$都连续.
\end{proposition}
