\section{扩充直线}

\begin{definition}{}{}
	记$\overline{\R}=\R\sqcup\left\{-\infty,+\infty\right\}$,其中$-\infty,+\infty\notin\R$.定义$\overline{\R}$上的全序$\leqslant$如下(暂时记$\R$上的全序为$\leqslant_{r}$):
	\begin{itemize}
		\item $\forall x\in\overline{\R}\left(-\infty<x<+\infty\right)$;
		\item $-\infty<+\infty$,
	\end{itemize}
	则$\forall x,y\in\R\left(x\leqslant_{r}y\iff x\leqslant y\right)$.记$\mathcal{B}=\left\{\left(a,b\right)\subseteq\overline{\R}\middle|a,b\in\overline{\R}\right\}$,给$\overline{\R}$配备以$\mathcal{B}$为拓扑基的拓扑,则该拓扑在$\R$上诱导的子空间拓扑和$\R$上的拓扑相同.
	
	我们称配备了以$\mathcal{B}$为基的拓扑的$\overline{\R}$为扩充实直线.
\end{definition}

\begin{proposition}{}{}
	设$a,b\in\R$且$a<b$,$f:\R\to\left(a,b\right)$是同胚,则存在唯一的同胚$\bar{f}:\overline{\R}\to\left[a,b\right]$使得$\bar{f}|_{\R}=f$.
	
	特别地,若$f$严格单调递增,则$f$是序同构.
\end{proposition}

\begin{proposition}{}{}
	$\overline{\R}$是\text{Hausdorff}紧空间.
\end{proposition}

\begin{corollary}{}{}
	$\overline{\R}$上存在唯一的一致结构$\mathcal{U}$与$\overline{\R}$上的拓扑相容,$\mathcal{U}$在$\R$上诱导的一致结构与$\R$的加法一致结构不相等.
\end{corollary}

\begin{proposition}{}{}
	$\overline{\R}$按$\leqslant$构成完备格.
\end{proposition}

\begin{proposition}{}{}
	设$A\subseteq\overline{\R}$,则$A$是$\overline{\R}$的连通集$\iff$ $A$是$\overline{\R}$的区间.
\end{proposition}

\begin{corollary}{}{}
	$\overline{\R}$是连通且局部连通的拓扑空间.
\end{corollary}

\begin{proposition}{}{}
	设$I$是$\overline{\R}$的区间,$f:I\to\overline{\R}$连续且严格单调,则$f$的余限制是同胚,并且$\im\left(f\right)$是$\overline{\R}$的区间.
\end{proposition}

\begin{proposition}{}{}
	$f:\overline{\R}\times\overline{\R}\to\overline{\R},\left(x,y\right)\mapsto\sup\left(x,y\right)$和$g:\overline{\R}\times\overline{\R}\to\overline{\R},\left(x,y\right)\mapsto\inf\left(x,y\right)$连续.
\end{proposition}
