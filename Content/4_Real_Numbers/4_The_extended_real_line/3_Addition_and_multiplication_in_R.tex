\section{\texorpdfstring{$\R$}{R}中的加法和乘法}

本小节中,记$A^{\prime}=\R\sqcup\left\{+\infty\right\}$,$A^{\prime\prime}=\R\sqcup\left\{-\infty\right\}$.

\begin{proposition}{}{}
	$-\id_{\R}$可以唯一地延拓为$\overline{\R}$上的连续子同胚$\iota$,并且满足$\iota\left(-\infty\right)=+\infty$,$\iota\left(+\infty\right)=-\infty$.
\end{proposition}

\begin{definition}{}{}
	按照如下方式将$\R$上的加法分别延拓为$A^{\prime}$和$A^{\prime\prime}$上的加法:
	\begin{itemize}
		\item 对任一$x\in A^{\prime}$,$x+\left(+\infty\right)=\left(+\infty\right)+x=+\infty$;
		\item 对任一$x\in A^{\prime\prime}$,$x+\left(-\infty\right)=\left(-\infty\right)+x=-\infty$.
	\end{itemize}
	容易验证,按上述延拓得到的映射只在$\left(-\infty,+\infty\right)$和$\left(+\infty,-\infty\right)$处不连续.
\end{definition}

\begin{proposition}{}{}
	在延拓后的加法下,$A^{\prime}$和$A^{\prime\prime}$各自构成加法交换半群,二者的零元都是$0$.
\end{proposition}

\begin{proposition}{}{}
	在延拓后的加法下,$A^{\prime}$中的不可消去元有且仅有$+\infty$,$A^{\prime\prime}$中的不可消去元有且仅有$-\infty$.
\end{proposition}

\begin{proposition}{}{}
	设$x,y,u,v\in\overline{\R}$且$x+y$和$u+v$有定义.
	\begin{enumerate}
		\item 若$x\leqslant y$且$u\leqslant v$,则$x+u\leqslant u+v$.
		\item 若$x<y$且$u<v$,则$x+u<u+v$.
	\end{enumerate}
\end{proposition}

\begin{definition}{}{}
	对任一$x\in\overline{\R}$,定义$x^{+}=\sup\left(x,0\right)$,$x^{-}=\sup\left(-x,0\right)$,$\left|x\right|=\sup\left(x,-x\right)$.
\end{definition}

\begin{proposition}{}{}
	$\left(+\infty\right)^{+}=+\infty$,$\left(-\infty\right)^{-}=+\infty$,$\left(+\infty\right)^{-}=0$,$\left(-\infty\right)^{+}=0$,$\left|+\infty\right|=\left|-\infty\right|=+\infty$.
\end{proposition}

\begin{proposition}{}{}
	对任一$x\in\overline{\R}$,有$x=x^{+}-x^{-}$,$\left|x\right|=x^{+}+x^{-}$.
\end{proposition}

\begin{proposition}{}{}
	设$x,y\in\overline{\R}$且$x+y$有定义,则$\left|x+y\right|\leqslant\left|x\right|+\left|y\right|$.
\end{proposition}

\begin{definition}{}{}
	按照如下方式将$\R$上的乘法延拓为$\overline{\R}\setminus\left\{0\right\}$上的乘法:
	\begin{itemize}
		\item 对任一$x\in\R_{>0}$,$x\cdot\left(+\infty\right)=\left(+\infty\right)\cdot x=+\infty$,$x\cdot\left(-\infty\right)=\left(-\infty\right)\cdot x=-\infty$;
		\item 对任一$x\in-\R_{>0}$,$x\cdot\left(+\infty\right)=\left(+\infty\right)\cdot x=-\infty$,$x\cdot\left(-\infty\right)=\left(-\infty\right)\cdot x=+\infty$.
	\end{itemize}
	容易验证,按上述延拓得到的映射只在$\left(0,+\infty\right)$,$\left(+\infty,0\right)$,$\left(0,-\infty\right)$,$\left(-\infty,0\right)$处不连续.
\end{definition}

\begin{proposition}{}{}
	在延拓后的乘法下,$\overline{\R}\setminus\left\{0\right\}$构成乘法交换半群,幺元为$1$,$+\infty$和$-\infty$在该乘法下不可消去.
\end{proposition}

\begin{proposition}{}{}
	设$x,y,z\in\overline{\R}$且$xz$和$yz$有定义.
	\begin{enumerate}
		\item 若$x\leqslant y$且$z>0$,则$xz\leqslant yz$.
		\item 若$x\leqslant y$且$z<0$,则$xz\geqslant yz$.
		\item 若$xy$有定义,则$\left|xy\right|=\left|x\right|\left|y\right|$.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	设$x,y,z\in\overline{\R}$且$x\left(y+z\right)$,$xz$和$yz$有定义,则$x\left(y+z\right)=xz+yz$.
\end{proposition}
