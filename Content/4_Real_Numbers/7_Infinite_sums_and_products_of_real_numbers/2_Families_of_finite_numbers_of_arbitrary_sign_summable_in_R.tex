\section{\texorpdfstring{$\R$}{R}中任意符号的有限可和族}

未加声明时,$\left(x_{i}\right)_{i\in I}$,$\left(y_{i}\right)_{i\in I}$是$\R$的元素族.

\begin{theorem}{}{}
	$\left(x_{i}\right)_{i\in I}$在$\R$中可和$\iff$ $\left(\left|x_{i}\right|\right)_{i\in I}$在$\R$中可和$\iff$ $\left(s_{H}\right)_{H\in\fin\left(I\right)}$在$\R$中有界.
\end{theorem}

\begin{theorem}{}{}
	设$\left(x_{i}\right)_{i\in I}$和$\left(y_{i}\right)_{i\in I}$可和,$\lambda,\mu\in\R$,则$\left(\lambda x_{i}+\mu y_{i}\right)_{i\in I}$可和且$\sum\limits_{i\in I}\left(\lambda x_{i}+\mu y_{i}\right)=\lambda\sum\limits_{i\in I}x_{i}+\mu\sum\limits_{i\in I}y_{i}$.
\end{theorem}
