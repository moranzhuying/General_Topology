\section{\texorpdfstring{$\R$}{R}中任意符号的有限可和族}

\begin{theorem}{}{}
	设$\left(x_{i}\right)_{i\in I}$是$\R$的元素族,对每个$H\in\fin\left(I\right)$记$s_{H}=\sum\limits_{i\in H}x_{i}$.如下陈述等价:
	\begin{enumerate}
		\item $\left(x_{i}\right)_{i\in I}$在$\R$中可和.
		\item $\left(\left|x_{i}\right|\right)_{i\in I}$在$\R$中可和.
		\item $\left(s_{H}\right)_{H\in\fin\left(I\right)}$在$\R$中有界.
	\end{enumerate}
\end{theorem}
