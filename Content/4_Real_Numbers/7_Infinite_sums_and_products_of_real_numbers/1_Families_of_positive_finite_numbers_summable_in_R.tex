\chapter{\texorpdfstring{$\R$}{R}中的无限和与无限乘积}

\section{\texorpdfstring{$\R$}{R}中的非负有限可和族}

记号约定:$\left(x_{i}\right)_{i\in I},\left(y_{i}\right)_{i\in I}$是$\R_{\geq 0}$的元素族.

\begin{theorem}{}{}
	对每个$H\in\fin\left(I\right)$记$s_{H}=\sum\limits_{i\in H}x_{i}$.下列陈述等价:
	\begin{enumerate}
		\item $\left(x_{i}\right)_{i\in I}$在$\R$中可和.
		\item $\left(s_{H}\right)_{H\in\fin\left(I\right)}$在$\R$中有上界.
	\end{enumerate}
	两条件之一成立时,$\sum\limits_{i\in I}x_{i}=\sup\limits_{H\in\fin\left(I\right)}s_{H}$.
\end{theorem}

\begin{remark}{}{}
	\begin{enumerate}
		\item 设$\left(I_{\lambda}\right)_{\lambda\in\Lambda}$是$\fin\left(I\right)$的元素族,满足$\forall H\in\fin\left(I\right)\exists\lambda\in\Lambda\left(H\subseteq I_{\lambda}\right)$,那么
		\begin{center}
			$\left(x_{i}\right)_{i\in I}$在$\R$中可和$\iff$ $\left(s_{I_{\lambda}}\right)_{\lambda\in\Lambda}$在$\R$中有上界.
		\end{center}
		\item 设$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$是$\R_{\geqslant 0}$的级数,那么
		\begin{center}	
			$\left(x_{n}\right)_{n=0}^{\infty}$在$\R$中可和$\iff$存在$\N$中的严格递增序列$\left(k_{n}\right)_{n=0}^{\infty}$使得$\left(s_{k_{n}}\right)_{n=0}^{\infty}$在$\R$中有上界.
		\end{center}
	\end{enumerate}
\end{remark}

\begin{example}{}{}
	\begin{enumerate}
		\item 对每个$q\in\left[0,1\right)\subseteq\R$,序列$\left(q^{n}\right)_{n=0}^{\infty}$在$\R$中可和,并且$\sum\limits_{n=0}^{\infty}q^{n}=\left(1-q\right)^{-1}$.
		\item 设$a,b\in\left[0,1\right)\subseteq\R$,那么$\left(a^{m}b^{n}\right)_{\left(m,n\right)\in\N\times\N}$在$\R$中可和.
		\item 对每个$p\geq 2$,序列$\left(n^{-p}\right)_{n=0}^{\infty}$在$\R$中可和.
		\item 序列$((n+1)^{-1})_{n=0}^{\infty}$在$\R$中不可和.
		\item 设$I$是$\R$中长度为$l\in\R_{\geq 0}$的有界区间,$\left(I_{n}\right)_{n=0}^{\infty}$是$\R$中一族两两不交且包含于$I$的开区间,那么$\left(\ell\left(I_{n}\right)\right)_{n=0}^{\infty}$可和且$\sum\limits_{n=0}^{\infty}\ell\left(I_{n}\right)\leq l$.
	\end{enumerate}
\end{example}

\begin{theorem}{可和族的比较判别法}{}
	设$\forall i\in I\left(x_{i}\leq y_{i}\right)$.
	\begin{enumerate}
		\item 如果$\left(y_{i}\right)_{i\in I}$在$\R$中可和,那么$\left(x_{i}\right)_{i\in I}$在$\R$中可和,$\sum\limits_{i\in I}x_{i}\leqslant\sum\limits_{i\in I}y_{i}$.
		\item 如果$\left(y_{i}\right)_{i\in I}$在$\R$中可和,并且$\exists k\in I\left(x_{k}<y_{k}\right)$,那么$\sum\limits_{i\in I}x_{i}<\sum\limits_{i\in I}y_{i}$.
	\end{enumerate}
\end{theorem}

\begin{corollary}{序列的比较判别法}{}
	设$\left(x_{n}\right)_{n=0}^{\infty},\left(y_{n}\right)_{n=0}^{\infty}$是$\R_{\geq 0}$的序列.
	\begin{enumerate}
		\item 如果$\left(y_{n}\right)_{n=0}^{\infty}$可和,并且$\exists a\in\R_{>0}\exists n_{0}\in\N\forall n\geq n_{0}\left(x_{n}\leqslant ay_{n}\right)$,那么$\left(x_{n}\right)_{n=0}^{\infty}$可和.
		\item 如果$\left(y_{n}\right)_{n=0}^{\infty}$不可和,并且$\exists a\in\R_{>0}\exists n_{0}\in\N\forall n\geq n_{0}\left(x_{n}\geqslant ay_{n}\right)$,那么$\left(x_{n}\right)_{n=0}^{\infty}$不可和.
	\end{enumerate}
\end{corollary}

\begin{example}{}{}
	\begin{enumerate}
		\item 设$a\in\R_{>0}$,那么$(a^{n}(n!)^{-1})_{n=0}^{\infty}$在$\R$中可和.
		\item 设$\left(a_{n}\right)_{n=0}^{\infty}$是$\R_{\geq 0}$中序列,并且在$\R$中可和,那么对每个$p\geq 2$,序列$\left(a_{n}^{p}\right)_{n=0}^{\infty}$在$\R$中可和.
		\item 设$a,b\in\left(-\infty,1\right)\subseteq\R$,那么$((a^{m}+b^{n})^{-1})_{\left(m,n\right)\in\N\times\N}$在$\R$中可和.
	\end{enumerate}
\end{example}

\begin{corollary}{}{}
	设$\left(x_{i}\right)_{i\in I}$在$\R$中可和.那么,对每个$J\subseteq I$,我们有$\sum\limits_{i\in J}x_{i}\leqslant\sum\limits_{i\in I}x_{i}$和$\sum\limits_{i\in J}x_{i}=\sum\limits_{i\in I}x_{i}\iff\forall i\in I\setminus J\left(x_{i}=0\right)$.
\end{corollary}
