\section{\texorpdfstring{$\R$}{R}中的无穷级数和无限乘积}

\begin{definition}{绝对收敛(级数)}{}
	设$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$是$\R$中的级数.如果$\left(\left(\left|x_{n}\right|\right)_{n=0}^{\infty},\left(\left|s_{n}\right|\right)_{n=0}^{\infty}\right)$在$\R$中收敛,则称$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$(在$\R$中)绝对收敛.
\end{definition}

\begin{proposition}{级数无条件收敛$\iff$级数绝对收敛$\iff$级数的通项可和}{}
	设$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$是$\R$中的级数.下列陈述等价:
	\begin{enumerate}
		\item $\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$无条件收敛.
		\item $\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$绝对收敛.
		\item $\left(x_{n}\right)_{n=0}^{\infty}$在$\R$中可和.
	\end{enumerate}
\end{proposition}

\begin{definition}{交错级数}{}
	设$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$是$\R$中的级数.如果存在$\R_{\geq 0}$中的序列$\left(v_{n}\right)_{n=0}^{\infty}$使得
	\begin{align*}
		\forall n\in\N\left(u_{n}=\left(-1\right)^{n}v_{n}\right),
	\end{align*}
	则称$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$是$\R$中的交错级数.
\end{definition}

\begin{example}{}{}
	考虑交错级数$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$,其中对每个$n\geq 0$有$x_{n}=\left(-1\right)^{n+1}\frac{1}{n+1}$,那么该级数绝对收敛,但是不收敛.
\end{example}

\begin{remark}{}{}
	如果$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(s_{n}\right)_{n=0}^{\infty}\right)$是交错级数,$\left(v_{n}\right)_{n=0}^{\infty}$单调递减且收敛到$0$,那么该交错级数收敛.
\end{remark}

\begin{definition}{绝对收敛(无穷乘积)}{}
	设$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(p_{n}\right)_{n=0}^{\infty}\right)$是$\R$中的无穷乘积.如果以$\left(1+\left|x_{n}\right|\right)_{n=0}^{\infty}$为通项的无穷乘积收敛,则称$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(p_{n}\right)_{n=0}^{\infty}\right)$绝对收敛.
\end{definition}

\begin{proposition}{}{}
	设$\left(\left(x_{n}\right)_{n=0}^{\infty},\left(p_{n}\right)_{n=0}^{\infty}\right)$是$\R$中的无穷乘积.下列陈述等价:
	\begin{enumerate}
		\item $\left(\left(x_{n}\right)_{n=0}^{\infty},\left(p_{n}\right)_{n=0}^{\infty}\right)$无条件收敛.
		\item $\left(\left(x_{n}\right)_{n=0}^{\infty},\left(p_{n}\right)_{n=0}^{\infty}\right)$绝对收敛.
		\item $\left(x_{n}\right)_{n=0}^{\infty}$绝对收敛.
	\end{enumerate}
\end{proposition}

\begin{example}{}{}
	定义$\R$中的序列$\left(x_{n}\right)_{n=0}^{\infty}$如下:对每个$n\geq 2$,
	\begin{align*}
		u_{2n-1}=-\frac{1}{n},u_{2n}=\frac{1}{n}.
	\end{align*}
	那么,以$\left(1+x_{n}\right)_{n=0}^{\infty}$为通项的无穷乘积不是绝对收敛的,但是收敛到$\frac{1}{2}$.
\end{example}
