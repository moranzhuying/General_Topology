\section{\texorpdfstring{$\R$}{R}中的可和族与可乘族}

\begin{proposition}{}{}
	\begin{enumerate}
		\item $\left[0,+\infty\right]$中的每个元素族在$\overline{\R}$中可和.
		\item $\left[-\infty,0\right]$中的每个元素族在$\overline{\R}$中可和.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	\begin{enumerate}
		\item 对$\left[0,+\infty\right]$中的每个可乘族$\left(x_{i}\right)_{i\in I}$,元素族$\left(1+x_{i}\right)_{i\in I}$在$\overline{\R}$中可乘.
		\item 对$\left[0,1\right]$中的每个可乘族$\left(x_{i}\right)_{i\in I}$,元素族$\left(1-x_{i}\right)_{i\in I}$在$\overline{\R}$中可乘.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	\begin{enumerate}
		\item 对$\left[0,+\infty\right]$中的每个可乘族$\left(x_{i}\right)_{i\in I}$,我们有$\prod\limits_{i\in I}\left(1+x_{i}\right)=+\infty\iff\sum\limits_{i\in I}x_{i}=+\infty$.
		\item 对$\left[0,1\right]$中的每个可乘族$\left(x_{i}\right)_{i\in I}$,我们有$\prod\limits_{i\in I}\left(1-x_{i}\right)=0\iff\sum\limits_{i\in I}x_{i}=0$.
	\end{enumerate}
\end{corollary}

\begin{corollary}{}{}
	设$\left(x_{i}\right)_{i\in I}$是$\R_{\geqslant 0}$的元素族.
	\begin{enumerate}
		\item $\prod\limits_{i\in I}x_{i}\in\R$ $\iff$ $\sum\limits_{i\in I}x_{i}$收敛.
		\item 设$\forall i\in I\left(x_{i}\leqslant 1\right)$,则$\prod\limits_{i\in I}\left(1-x_{i}\right)\in\R$ $\iff$ $\sum\limits_{i\in I}x_{i}<+\infty$.
		\item 若$\sum\limits_{i\in I}x_{i}=+\infty$,则$\prod\limits_{i\in I}\left(1+x_{i}\right)=+\infty$,$\prod\limits_{i\in I}\left(1-x_{i}\right)=0$.
	\end{enumerate}
\end{corollary}

\begin{proposition}{}{}
	设$\left(I_{\lambda}\right)_{\lambda\in\Lambda}$是集合$I$的划分.
	\begin{enumerate}
		\item 如果$\left(x_{i}\right)_{i\in I}$是$\left[0,+\infty\right]$的可和族,那么$\sum\limits_{i\in I}x_{i}=\sum\limits_{\lambda\in\Lambda}\sum\limits_{i\in I_{\lambda}}x_{i}$.
		\item 如果$\left(x_{i}\right)_{i\in I}$是$\left[0,1\right]$的可乘族(相对的,$\left[1,+\infty\right]$的可乘族),那么$\prod\limits_{i\in I}x_{i}=\prod\limits_{\lambda\in\Lambda}\prod\limits_{i\in I_{\lambda}}x_{i}$.
	\end{enumerate}
\end{proposition}
