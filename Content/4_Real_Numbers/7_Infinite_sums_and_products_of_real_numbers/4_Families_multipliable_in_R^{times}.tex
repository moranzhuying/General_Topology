\section{\texorpdfstring{$\R^{\times}$}{R*}中的可乘族}

\begin{lemma}{Weierstrass不等式}{}
	设$n\geq 1$,$\left(x_{i}\right)_{i=1}^{n}$是$\R$中的序列.
	\begin{enumerate}
		\item 若$\forall 1\leq i\leq n\left(x_{i}>0\right)$,那么$\prod\limits_{i=1}^{n}\left(1+x_{i}\right)\geq 1+\sum\limits_{i=1}^{n}x_{i}$.
		\item 若$\forall 1\leq i\leq n\left(0<x_{i}<1\right)$,那么$\prod\limits_{i=1}^{n}\left(1-x_{i}\right)\geq 1-\sum\limits_{i=1}^{p}x_{i}$.
	\end{enumerate}
\end{lemma}

\begin{theorem}{}{}
	设$\left(x_{i}\right)_{i\in I}$是$\left(-1,+\infty\right)$的元素族,那么$\left(1+x_{i}\right)_{i\in I}$在$\R^{\times}$中可乘$\iff$ $\left(x_{i}\right)_{i\in I}$在$\R$中可和.
\end{theorem}
