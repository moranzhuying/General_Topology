\section{两个无限和的乘积}

\begin{proposition}{}{}
	如果$\left(x_{i}\right)_{i\in I},\left(y_{j}\right)_{j\in J}$是$\R$中的可和族,那么$\left(x_{i}y_{j}\right)_{\left(i,j\right)\in I\times J}$是$\R$中的可和族,并且
	\begin{align*}
		\sum\limits_{\left(i,j\right)\in I\times J}x_{i}y_{j}=\left(\sum\limits_{i\in I}x_{i}\right)\left(\sum\limits_{j\in J}y_{j}\right)
	\end{align*}
\end{proposition}
