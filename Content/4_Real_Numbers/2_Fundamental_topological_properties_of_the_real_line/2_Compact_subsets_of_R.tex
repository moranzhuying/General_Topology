\section{\texorpdfstring{$\R$}{R}的紧集}

记号约定:$A$是实直线$\R$的子集.

\begin{theorem}{Heine-Borel-Lebesgue定理}{}
	$A$是$\R$的紧集$\iff$ $A$是$\R$的有界闭集.
\end{theorem}

\begin{corollary}{$\R$中相对紧$\iff$有界}{}
	$A$是$\R$的相对紧集$\iff$ $A$按$\leqslant$有界.
\end{corollary}

\begin{corollary}{$\R$局部紧但不紧}{}
	实直线是局部紧空间,但不是紧空间.
\end{corollary}
