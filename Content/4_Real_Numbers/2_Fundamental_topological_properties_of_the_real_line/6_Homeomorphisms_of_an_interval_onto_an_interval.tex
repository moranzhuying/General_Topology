\section{区间之间的同胚}

记号约定:$I$是$\R$中的区间,$f\in\Hom_{\mathsf{Set}}\left(I,\R\right)$.

\begin{theorem}{}{}
	$f^{\circ}$是同胚$\iff$ $f\in\Hom_{\mathsf{Top}}\left(I,\R\right)$且严格保序.两条件之一成立时,$\im\left(f\right)$是$\R$的区间.
\end{theorem}

\begin{proposition}{}{}
	如果$f$连续且单调,那么$f$严格保序,$f^{\circ}$是同胚.
\end{proposition}
