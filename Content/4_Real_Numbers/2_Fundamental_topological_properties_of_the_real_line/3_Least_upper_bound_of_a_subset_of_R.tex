\section{\texorpdfstring{$\R$}{R}中子集的上确界}

\begin{theorem}{$\R$中的确界原理}{}
	有序群$\R$是条件完备格.
\end{theorem}

\begin{proposition}{上确界的特征刻画}{}
	设$A$是$\R$的非空上有界子集,$c\in\R$,那么$c=\sup\left(A\right)\iff A^{u}\cap A^{lu}=\left\{c\right\}$.
\end{proposition}

\begin{proposition}{闭集的最大元和闭包的最大元}{}
	设$A$是$\R$的非空上有界子集.
	\begin{enumerate}
		\item 如果$A$是闭集,那么$\sup\left(A\right)=\max\left(A\right)\in A$.
		\item 设$c\in\R$,则$c=\sup\left(A\right)\iff c=\max\left(\closure\left(A\right)\right)$.
	\end{enumerate}
\end{proposition}
