\section{实数在基列下的展开式}

\begin{lemma}{}{}
	对任一$x\in\R$,存在唯一的$n\in\Z$使得$n\leqslant x<n+1$且$n=\max\left\{k\in\Z\middle|k\leqslant x\right\}$.
\end{lemma}

\begin{lemma}{}{}
	对任一$x\in\R$,存在唯一的$n\in\Z$使得$n<x-1\leqslant n$且$n=\min\left\{k\in\Z\middle|k\geqslant x\right\}$.
\end{lemma}

\begin{definition}{}{}
	定义$\lfloor\cdot\rfloor:\R\to\Z,x\mapsto\max\left\{n\in\Z\middle|n\leqslant x\right\}$,$\lceil\cdot\rceil:\R\to\Z,x\mapsto\min\left\{n\in\Z\middle|n\geqslant x\right\}$.
	\begin{itemize}
		\item $\lfloor\cdot\rfloor$和$\lceil\cdot\rceil$分别称为下取整函数和上取整函数.
		\item 对任一$x\in\R$,分别称$\lfloor x\rfloor$和$\lceil x\rceil$为$x$的上取整部分(整数部分)和下取整部分.
	\end{itemize}
\end{definition}

\begin{definition}{}{}
	设$\left(a_{n}\right)_{n=0}^{\infty}$和$\left(d_{n}\right)_{n=0}^{\infty}$是$\Z_{\geqslant 0}$中的序列,并且满足如下条件:
	\begin{itemize}
		\item $\forall n\in\Z_{\geqslant 0}\left(d_{n+1}=a_{n+1}d_{n}\right)$;
		\item $\left(d_{n}\right)_{n=0}^{\infty}$严格单调递增;
		\item $d_{0}=1$.
	\end{itemize}
	可以验证$\forall n\in\Z_{\geqslant 0}\left(a_{n}\geqslant 2\right)$.我们称$\left(d_{n}\right)_{n=0}^{\infty}$是基序列,$\left(a_{n}\right)_{n=0}^{\infty}$是底因子列.
\end{definition}

\begin{definition}{}{}
	设$\left(d_{n}\right)_{n=0}^{\infty}$是底序列,$x\in\R$.$\Q$中的序列$\left(\frac{\lfloor xd_{n}\rfloor}{d_{n}}\right)_{n=0}^{\infty}$称为$x$关于$\left(d_{n}\right)_{n=0}^{\infty}$的不足近似列.
\end{definition}

\begin{proposition}{}{}
	设$x\in\R$,$\left(d_{n}\right)_{n=0}^{\infty}$,$\left(a_{n}\right)_{n=0}^{\infty}$,$\left(r_{n}\right)_{n=0}^{\infty}$依次是底序列,底因子列,$x$关于$\left(d_{n}\right)_{n=0}^{\infty}$的不足近似列.
	
	那么,$\Z$中满足下列条件的序列$\left(u_{n}\right)_{n=1}^{\infty}$唯一:
	\begin{itemize}
		\item $\forall n\in\Z_{\geqslant 1}\left(0\leqslant u_{n}\leqslant a_{n}-1\wedge \lfloor xd_{n}\rfloor=a_{n}\lfloor xd_{n-1}\rfloor+u_{n}\right)$;
		\item $\forall n\in\Z_{\geqslant 1}\left(r_{n}=\lfloor x\rfloor+\sum\limits_{k=1}^{n}\frac{u_{k}}{d_{k}}\right)$.
	\end{itemize}
\end{proposition}

\begin{theorem}{}{}
	对任一$x\in\R$,有$x=\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{d_{n}}$,其中:
	\begin{itemize}
		\item $\left(d_{n}\right)_{n=0}^{\infty}$,$\left(a_{n}\right)_{n=0}^{\infty}$,$\left(r_{n}\right)_{n=0}^{\infty}$依次是底序列,底因子列,$x$关于$\left(d_{n}\right)_{n=0}^{\infty}$的不足近似列.
		\item $\left(u_{n}\right)_{n=0}^{\infty}$是$\Z$中的序列,满足$\forall n\in\Z_{\geqslant 1}\left(0\leqslant u_{n}\leqslant a_{n}-1\wedge \lfloor xd_{n}\rfloor=a_{n}\lfloor xd_{n-1}\rfloor+u_{n}\wedge r_{n}=\lfloor x\rfloor+\sum\limits_{k=1}^{n}\frac{u_{k}}{d_{k}}\right)$.
		\item $\sum\limits_{n=1}^{\infty}\frac{u_{n}}{d_{n}}$收敛.
	\end{itemize}
	我们称$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{d_{n}}$是$x$关于$\left(d_{n}\right)_{n=0}^{\infty}$的展开式,对任一$n\in\Z_{\geqslant 1}$,$u_{n}$称为$x$的第$n$位数字.
\end{theorem}
