\section{实数的\texorpdfstring{$a$}{a}进展开式}

\begin{definition}{}{}
	设$p\in\Z_{\geqslant 2}$.我们称$x\in\R$关于$\left(p^{n}\right)_{n=0}^{\infty}$的展开式$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{p^{n}}$是$x$的$p$进制展开式.特别地:
	\begin{itemize}
		\item $p=10$时,称$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{p^{n}}$是$x$的十进制展开式.
		\item $p=3$时,称$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{p^{n}}$是$x$的三进制展开式.
		\item $p=2$时,称$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{p^{n}}$是$x$的二进制展开式.
	\end{itemize}
\end{definition}

\begin{convention}{}{}
	设$p\in\Z_{\geqslant 2}$,$x\in\R_{\geqslant 0}$的$p$进制展开式为$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{p^{n}}$.对任一$n\in\Z_{\geqslant 1}$,记$\left(\overline{\lfloor x\rfloor.u_{1}\ldots u_{n}}\right)_{p}=u_{0}+\sum\limits_{k=1}^{n}u_{k}$.
\end{convention}
