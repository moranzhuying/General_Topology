\section{展开式的比较}

未加声明时,$\left(d_{n}\right)_{n=0}^{\infty}$是底序列,$\left(a_{n}\right)_{n=0}^{\infty}$是底因子列,$x,y\in\R$各自在$\left(d_{n}\right)_{n=0}^{\infty}$下的正则展开式依次是
\begin{align*}
	\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{d_{n}},\lfloor y\rfloor+\sum\limits_{n=1}^{\infty}\frac{v_{n}}{d_{n}}.
\end{align*}

\begin{proposition}{}{}
	设$x\neq y$.
	\begin{enumerate}
		\item 若$\lfloor x\rfloor<\lfloor y\rfloor$,则$x<y$.
		\item 若$\lfloor x\rfloor=\lfloor y\rfloor\wedge \exists m\in\Z_{\geqslant 1}\left(\forall n\in\llbracket 1,m-1\rrbracket\left(u_{n}=v_{n}\right)\wedge u_{m}<v_{m}\right)$,则$x<y$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	设$\lfloor x\rfloor=\lfloor y\rfloor\wedge \exists m\in\Z_{\geqslant 1}\left(\forall n\in\llbracket 1,m-1\rrbracket\left(u_{n}=v_{n}\right)\wedge u_{m}<v_{m}\right)$.
	\begin{enumerate}
		\item $\left|x-y\right|\leqslant\frac{1}{d_{m}}$.
		\item 对$\left[x,y\right]$中任意有正常展开式$\lfloor z\rfloor+\sum\limits_{n=1}^{\infty}\frac{w_{n}}{d_{n}}$的元素$z$,有$\lfloor z\rfloor=\lfloor y\rfloor\wedge \exists m\in\Z_{\geqslant 1}\forall n\in\llbracket 1,m-1\rrbracket\left(w_{n}=v_{n}\right)$
	\end{enumerate}
\end{corollary}

\begin{corollary}{}{}
	设$X=\Z\times\prod\limits_{n=1}^{\infty}\llbracket 0,a_{n}-1\rrbracket$,给$\Z$和$\llbracket 0,a_{n}-1\rrbracket$($n\in\Z_{\geqslant 1}$)配备离散拓扑,$X$配备乘积拓扑.定义
	\begin{align*}
		\varphi:X\to\R,\left(p_{0},\left(u_{n}\right)_{n=0}^{\infty}\right)\mapsto p_{0}+\sum\limits_{n=1}^{\infty}\frac{u_{n}}{d_{n}},
	\end{align*}
	则$\varphi$是连续映射.
\end{corollary}
