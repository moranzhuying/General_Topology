\section{Cantor级数展开式的比较}

记号约定:
\begin{itemize}
	\item $\left(d_{n}\right)_{n=0}^{\infty}$是基列,$\left(a_{n}\right)_{n=0}^{\infty}$是底列.
	\item $x,y\in\R$,各自在$\left(d_{n}\right)_{n=0}^{\infty}$下的Cantor级数展开式为$\lfloor x\rfloor+\sum\limits_{n=1}^{\infty}\dfrac{u_{n+1}}{d_{n+1}}$和$\lfloor y\rfloor+\sum\limits_{n=0}^{\infty}\dfrac{v_{n+1}}{d_{n+1}}$.
\end{itemize}

\begin{proposition}{}{}
	下列条件之一成立时,我们有$x<y$.
	\begin{enumerate}
		\item $\lfloor x\rfloor<\lfloor y\rfloor$.
		\item $\lfloor x\rfloor=\lfloor y\rfloor$,并且$\exists m\geq 1\left(\forall 1\leq i\leq m-1\left(u_{n}=v_{n}\right)\wedge u_{m}<v_{m}\right)$.
	\end{enumerate}
\end{proposition}

\begin{proposition}{}{}
	设$\lfloor x\rfloor=\lfloor y\rfloor$,$m\geq 1$且$\forall 1\leq i\leq m-1\left(u_{i}=v_{i}\right)\wedge u_{m}<v_{m}$,那么
	\begin{enumerate}
		\item $\left|x-y\right|\leq\dfrac{1}{d_{m}}$.
		\item 对每个$z\in\left[x,y\right]$,如果$z$在$\left(d_{n}\right)_{n=0}^{\infty}$下的Cantor级数展开式为$\lfloor z\rfloor+\sum\limits_{n=0}^{\infty}\dfrac{w_{n+1}}{d_{n+1}}$,那么
		\begin{align*}
			\lfloor z\rfloor=\lfloor y\rfloor\wedge\forall 1\leq i\leq m-1\left(v_{n}=w_{n}\right).
		\end{align*}
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	沿用\cref{evaluation-map-over-cantor-number-space}的记号.给$\Z$和$\left(I_{n}\right)_{n=1}^{\infty}$中的每一项配备离散拓扑,$X$配备乘积拓扑,那么$\varphi$连续.
\end{corollary}
