\section{\texorpdfstring{$\R$}{R}的基数}

\begin{lemma}{}{}
	$\left|\R\right|=\left|\left[0,1\right]\right|$.
\end{lemma}

\begin{theorem}{}{}
	$\left|\left[0,1\right]\right|=\left|\mathcal{P}\left(\N\right)\right|$.
\end{theorem}

\begin{corollary}{}{}
	$\left|\R\right|=2^{\aleph_{0}}$.
\end{corollary}

\begin{corollary}{}{}
	对任一可数集$I$,有$\left|\R\right|>\left|X\right|$.
\end{corollary}

\begin{corollary}{}{}
	对$\R$中每个不是单点集或空集的区间$I$,有$\left|I\right|=\left|\R\right|$.特别地,$\left|\R\setminus\Q\right|=\left|\R\right|$.
\end{corollary}
